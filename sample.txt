  Future<void> getPatientDetail() async {
    setState(() {
      loading = true;
    });

    await profileService.getProfileDetail();
    if (profileService.response_Status_Code_API_1 == 200) {
      setState(() {
        profileModel = profileService.profileResponseModel;
        firstNameTextEditingController.text = profileModel!.firstName!;
        lastNameTextEditingController.text = profileModel!.lastName!;
        lastPasswordChangeDate = profileModel!.lastChangePasswordDate!;
        idPhotoBase64 = profileModel!.userIdPhotoBase64;
        profilePhotoBase64 = profileModel!.userProfilePhotoBase64;
        if (profilePhotoBase64!.isNotEmpty) {
          GlobalParams.USER_PROFILE_PHOTO_BASE64 =
              Base64Decoder().convert(profilePhotoBase64!);
        } else {
          GlobalParams.USER_PROFILE_PHOTO_BASE64 = null;
        }
        GlobalParams.USER_ID_PHOTO_BASE64 = profileModel!.userIdPhotoBase64!;
        GlobalParams.USER_INITIAL = profileModel!.userInitial!;
        tempFirstName = profileModel!.firstName!;
        tempLastName = profileModel!.lastName!;

        loading = false;
      });
    } else {
      if (profileService.response_Status_Code_API_1 != 205) {
        setState(() {
          loading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  Future<void> updatePatientDetail() async {
    if (profileModel!.userProfilePhotoBase64!.isNotEmpty) {

    } else {
      GlobalParams.USER_PROFILE_PHOTO_BASE64 = null;

    GlobalParams.USER_ID_PHOTO_BASE64 = profileModel!.userIdPhotoBase64!;
    profileModel!.isProfilePhotoDeleting = isProfilePhotoDeleting;
    profileModel!.isIdPhotoDeleting = isIdPhotoDeleting;
    profileModel!.isProfilePhotoUploading = isProfilePhotoUploading;
    profileModel!.isIdPhotoUploading = isIdPhotoUploading;
    profileModel!.contentType = contentType;

    if (tempFirstName != firstNameTextEditingController.text || //MBT 47776
        tempLastName != lastNameTextEditingController.text) {
      profileModel!.isNameChanged = true;
      profileModel!.firstName = firstNameTextEditingController.text;
      profileModel!.lastName = lastNameTextEditingController.text;
    } else {
      profileModel!.isNameChanged = false;
    }

    await profileService.saveProfileDetail(profileModel!);
    if (profileService.response_Status_Code_API_2 == 200 &&
        profileService.profileResponse.toLowerCase() == 'true') {
      setState(() {
        GlobalParams.USER_ID_PIC = GlobalParams.USER_ID_PHOTO_BASE64;
        isIdPhotoDeleting = false;
        isProfilePhotoDeleting = false;
        isProfilePhotoUploading = false;
        isIdPhotoUploading = false;

          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }

  Future<String> getBase64Image(XFile? image) async {
    List<int> imageBytes = await File(image!.path).readAsBytesSync();
    String imgBase64 = base64Encode(imageBytes);
    return imgBase64;
  }

  Future<String> getBase64ImageFromCrop(CroppedFile? image) async {
    List<int> imageBytes = await File(image!.path).readAsBytesSync();
    String imgBase64 = base64Encode(imageBytes);
    return imgBase64;
  }

  Future<CroppedFile?> _cropImage(File imageFile) async {
    return await ImageCropper().cropImage(
      sourcePath: imageFile.path,
      uiSettings: [
        AndroidUiSettings(
            toolbarTitle: 'Crop Image',
            toolbarColor: Colors.deepOrange,
            toolbarWidgetColor: Colors.white,
            initAspectRatio: CropAspectRatioPreset.original,
            lockAspectRatio: false),
        IOSUiSettings(
          minimumAspectRatio: 1.0,
        ),
      ],
    );
  }

  checkFileSize(PlatformFile file) {
    final size = file.size / 1024;
    if (profilePhoto) {
      if (size <= 2048) {
        return true;
      }
    } else {
      if (size <= 2048) {
        return true;
      }
    }
    return false;
  }

  checkFileFormat(PlatformFile file) {
    final extension = file.extension;

    if (AllowedExtensions.contains(extension)) {
      if (extension == "jpg" ||
          extension == "JPG" ||
          extension == "jpeg" ||
          extension == "JPEG") {
        contentType = "jpg";
      } else {
        contentType = "png";
      }
      return true;
    }
    return false;
  }

  convertFileToBase64(PlatformFile file) {
    setState(() async {
      final pickedFile = file.bytes;
      if (pickedFile != null) {
        var file64 = base64Encode(pickedFile);
        if (profilePhoto) {
          profilePhotoBase64 = file64;
          profileModel!.userProfilePhotoBase64 = profilePhotoBase64;
          isProfilePhotoUploading = true;
        } else {
          idPhotoBase64 = file64;
          profileModel!.userIdPhotoBase64 = idPhotoBase64;
          isIdPhotoUploading = true;
          isIdPhotoDeleting = false;
        }
      }
    });
    updatePatientDetail();
  }

  void loadSelectedFiles(PlatformFile file) {
    if (checkFileSize(file) && checkFileFormat(file)) {
      convertFileToBase64(file);
    } else {
      if (!checkFileFormat(file)) {
        showCustomDialog(
          context,
          title:
              "Invalid Image Format! Select an alternative image source file such as a PNG, JPEG or JPG",
          firstButtonText: "Ok",
          secondButtonText: "",
          iconData: Symbols.warning,
          firstButtonFunction: () {
            Get.back();
          },
        );
      } else {
        showToast(
          context,
          message: profilePhoto
              ? 'The attachment size exceeds the allowable limit. Maximum size of attachment allowed is 2 MB.'
              : 'The attachment size exceeds the allowable limit. Maximum size of attachment allowed is 2 MB.',
          type: ToastificationType.warning,
        );
      }
    }
  }
