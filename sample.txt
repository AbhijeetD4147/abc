  Future<void> getHealthSummaryList(DateRange? dateRange) async {
    setState(() {
      selectedHealthSummaryId = 0;
      healthSummaryListLoading = true;
      healthSummaryListModel.clear();
    });
    await healthSummaryService.getHealthSummaryList(
      currentPage,
      dateFormat.format(dateRange!.start),
      dateFormat.format(dateRange.end),
    );

    if (healthSummaryService.response_Status_Code_API_1 == 200) {
      setState(() {
        healthSummaryListModel = healthSummaryService.healthSummaryListModel!;
        if (healthSummaryListModel.length > 0) {
          var totalRecords = healthSummaryListModel[0].totalRecords;
          totalPages = (totalRecords! / 10).ceil();
          lastPage = totalPages;
          healthSummaryListModel[0].isSeen = true;
          selectedHealthSummaryId = healthSummaryListModel[0].healthSummaryId!;
          selectedHealthSummaryIndex = 0;
          if (Responsive.isMobile(context)) {
            healthSummaryListLoading = false;
          } else {
            isHealthSummaryScreenVisibile = true;
            getHealthSummaryDetails(selectedHealthSummaryId);
          }
        } else {
          if (currentPage > 1) {
            currentPage--;
            getHealthSummaryList(selectedDateRange);
          } else {
            selectedHealthSummaryId = 0;
            healthSummaryDataLoading = false;
            healthSummaryListLoading = false;
          }
        }
      });
    } else {
      if (healthSummaryService.response_Status_Code_API_1 != 205) {
        setState(() {
          healthSummaryListLoading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  Future<void> updateHealthSummaryAuditLog(String action) async {
    await activityLogService.updateAuditLogForHealthSummary(
        action,
        selectedHealthSummaryId,
        emailIdRequiredInPayloadForAuditLog ? toController.text.trim() : "");
    if (activityLogService.response_Status_Code_API_3 == 200) {
      toController.clear();
    } else if (activityLogService.response_Status_Code_API_3 != 200 &&
        activityLogService.response_Status_Code_API_3 != 205) {
      showToast(
        context,
        message:
            'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
        type: ToastificationType.error,
      );
    }
  }

  Future<void> getHealthSummaryDetails(int id) async {
    setState(() {
      healthSummaryDataLoading = true;
    });
    await healthSummaryService.getHealthSummaryThread(id);
    if (healthSummaryService.response_Status_Code_API_2 == 200) {
      setState(() {
        healthSummaryThreadModel =
            healthSummaryService.healthSummaryThreadModel;
        healthSummarySubject = healthSummaryThreadModel!.summaryTitle;
        healthSummaryData = healthSummaryThreadModel!.summaryData;
        healthSummaryThreadAttachmentList =
            healthSummaryThreadModel!.healthSummaryAttachment ?? [];
        healthSummaryDataLoading = false;
        healthSummaryListLoading = false;
      });
    } else {
      if (healthSummaryService.response_Status_Code_API_2 != 205) {
        setState(() {
          healthSummaryDataLoading = false;
        });

          context,
          message:

        );
      }
    }

  Future<void> sendMessage() async {
    if (subjectController.text.isNotEmpty &&
        toController.text.isNotEmpty &&
        !validateSubject &&
        !validateToEmail) {
      setState(() {
        healthSummaryForShareLoading = true;
      });

      DateFormat dateFormat = DateFormat("yyyy-MM-dd hh:mm");

      HealthSummaryTransmitModel healthSummaryTransmitModel =
          HealthSummaryTransmitModel();
      healthSummaryTransmitModel.ptcustomerId =
          int.parse(GlobalParams.PT_CUSTOMER_ID);
      healthSummaryTransmitModel.isActive = true;
      healthSummaryTransmitModel.toEmail = toController.text.trim();
      healthSummaryTransmitModel.messageSubject = subjectController.text.trim();
      healthSummaryTransmitModel.messageText = messageController.text.trim();
      healthSummaryTransmitModel.messageDateTime =
          dateFormat.format(currentDate);
      healthSummaryTransmitModel.messageStatus = "Sent";
      healthSummaryTransmitModel.isDirectMessage = false;
      healthSummaryTransmitModel.kno2Id = "";
      healthSummaryTransmitModel.isCustomer = true;
      healthSummaryTransmitModel.isIncoming = false;
      healthSummaryTransmitModel.isFavourite = true;
      healthSummaryThreadAttachmentList.removeAt(1);
      healthSummaryTransmitModel.ptMessageAttachment =
          healthSummaryThreadAttachmentList;
      healthSummaryTransmitModel.isReplied = false;
      healthSummaryTransmitModel.parentMessageId = 0;
      healthSummaryTransmitModel.ptPracticePersonId = 118;
      healthSummaryTransmitModel.userId = int.parse(GlobalParams.USER_ID);
      healthSummaryTransmitModel.switchUserId =
          int.parse(GlobalParams.SWITCH_USER_ID);

      await healthSummaryService
          .transmitHealthSummary(healthSummaryTransmitModel);
      if (healthSummaryService.response_Status_Code_API_3 == 200) {
        setState(() {
          emailIdRequiredInPayloadForAuditLog = true;
          isHealthSummaryForwardScreen = false;
          isMessageSentScreenVisible = true;
          healthSummaryForShareLoading = false;
          messageController.clear();
          subjectController.clear();
        });
        await updateHealthSummaryAuditLog("Transmit CCDA");
      } else {
        if (healthSummaryService.response_Status_Code_API_3 != 205) {
          setState(() {
            healthSummaryListLoading = false;
          });
          showToast(
            context,
            message:
                'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
            type: ToastificationType.error,
          );
        }
      }
    }
    if (subjectController.text.isEmpty) {
      setState(() {
        validateSubject = true;
      });
    }
    if (toController.text.isEmpty) {
      setState(() {
        validateToEmail = true;
        invalidEmailMessage = 'Email is required';
      });
    }
  }

  downloadFile(String? base64File, String? fileName) {
    final blob = html.Blob([base64Decode(base64File!)]);
    final url = html.Url.createObjectUrlFromBlob(blob);
    final anchor = html.document.createElement('a') as html.AnchorElement
      ..href = url
      ..style.display = 'none'
      ..download = fileName;
    html.document.body!.children.add(anchor);

    anchor.click();

    html.document.body!.children.remove(anchor);
    html.Url.revokeObjectUrl(url);
  }

  Future _pickDateRange(BuildContext context) async {
    pickedDateTimeRange = await showDateRangePicker(
      context: context,
      firstDate: DateTime(1950),
      lastDate: DateTime.now(),
      initialDateRange: dateTimeRange,
      saveText: 'OK',
      cancelText: 'CANCEL',
      initialEntryMode: DatePickerEntryMode.calendar,
      currentDate: DateTime.now(),
      fieldStartLabelText: 'From',
      fieldEndLabelText: 'To',
      fieldEndHintText: 'MM/DD/YYYY',
      fieldStartHintText: 'MM/DD/YYYY',
      keyboardType: TextInputType.number,
      builder: (context, child) {
        return Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Theme(
              data: Theme.of(context).copyWith(
                colorScheme: ColorScheme.light(
                  primary: ThemeSelection.BGColor, // header background color
                  onPrimary: Colors.white, // header text color
                  onSurface: Colors.black, // body text color
                ),
                textButtonTheme: TextButtonThemeData(
                  style: ButtonStyle(
                    foregroundColor: WidgetStateProperty.resolveWith<Color>(
                        (Set<WidgetState> states) {
                      if (states.contains(WidgetState.hovered))
                        return ThemeSelection
                            .primaryPressedBackgroundButtonColor;
                      if (states.contains(WidgetState.focused))
                        return ThemeSelection.whiteButtonHover;
                      return ThemeSelection.BGColor;
                    }),
                  ),
                ),
              ),
              child: ConstrainedBox(
                constraints: BoxConstraints(
                  maxWidth: Responsive.isMobile(context)
                      ? 90.w
                      : Responsive.isTablet(context)
                          ? 45.w
                          : 23.w,
                  maxHeight: Responsive.isMobile(context)
                      ? 70.h
                      : Responsive.isTablet(context)
                          ? 70.h
                          : 60.h,
                ),
                child: child,
              ),
            ),
          ],
        );
      },
    );

    if (pickedDateTimeRange != null) {
      setState(() {
        dateTimeRange = pickedDateTimeRange;
        DateRange dateRange =
            DateRange(dateTimeRange!.start, dateTimeRange!.end);
        dateController.text = '${dateFormat.format(dateRange.start)}' +
            " - " +
            '${dateFormat.format(dateRange.end)}';
        currentPage = 1;
        getHealthSummaryList(dateRange);
      });
    }
  }

  Widget datePickerBuilder(
      BuildContext context, dynamic Function(DateRange?) onDateRangeChanged,
      [bool doubleMonth = true]) {
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (mounted) {
        setState(() {
          isDatePickerOpen = true;
        });
      }
    });
    return DateRangePickerWidget(
      height: 50.h,
      theme: CalendarTheme(
        selectedColor: Colors.blue,
        dayNameTextStyle: TextStyle(color: Colors.black45, fontSize: 10),
        inRangeColor: Color(0xFFD9EDFA),
        inRangeTextStyle: TextStyle(color: Colors.blue),
        selectedTextStyle: TextStyle(color: Colors.white),
        todayTextStyle: TextStyle(fontWeight: FontWeight.bold),
        defaultTextStyle: TextStyle(color: Colors.black, fontSize: 12),
        radius: 10,
        tileSize: 40,
        disabledTextStyle: TextStyle(color: Colors.grey),
      ),
      doubleMonth: doubleMonth,
      quickDateRanges: [
        QuickDateRange(
          dateRange: DateRange(
            currentDate,
            currentDate,
          ),
          label: "Today",
        ),
        QuickDateRange(
          label: 'Yesterday',
          dateRange: DateRange(
            currentDate.subtract(const Duration(days: 1)),
            currentDate.subtract(const Duration(days: 1)),
          ),
        ),
        QuickDateRange(
          label: 'Last 7 days',
          dateRange: DateRange(
            currentDate.subtract(const Duration(days: 6)),
            currentDate,
          ),
        ),
        QuickDateRange(
          label: 'Last 30 days',
          dateRange: DateRange(
            currentDate.subtract(const Duration(days: 29)),
            currentDate,
          ),
        ),
        QuickDateRange(
          label: 'This Month',
          dateRange: DateRange(
            DateTime(currentDate.year, currentDate.month, 1),
            currentDate,
          ),
        ),
        QuickDateRange(
          label: 'Last Month',
          dateRange: DateRange(
            DateTime(currentDate.year, currentDate.month - 1, 1),
            DateTime(currentDate.year, currentDate.month, 0),
          ),
        ),
        QuickDateRange(
          label: 'This Year',
          dateRange: DateRange(
            DateTime(currentDate.year, 1, 1),
            currentDate,
          ),
        ),
      ],
      initialDateRange: selectedDateRange,
      initialDisplayedDate: selectedDateRange?.start ?? DateTime.now(),
      onDateRangeChanged: (value) {
        if (value != null) {
          setState(() {
            isDatePickerOpen = false;
          });
        }
        onDateRangeChanged(value);
      },
      maxDate: DateTime.now(),
    );
  }
}

class CustomFormattedDateRange extends DateRange {
  CustomFormattedDateRange(DateTime start, DateTime end) : super(start, end);

  @override
  String toString() {
    final DateFormat formatter = DateFormat('MM/dd/yyyy');
    return '${formatter.format(start)} - ${formatter.format(end)}';
  }
}
