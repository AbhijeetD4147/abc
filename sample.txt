class RecordMatchFoundScreen extends StatefulWidget {
  String alertMsg;
  String userMatchType;

  RecordMatchFoundScreen(
      {super.key, required this.alertMsg, required this.userMatchType});

  @override
  State<RecordMatchFoundScreen> createState() =>
      _PatientRecordMatchFoundScreenState();
}

class _PatientRecordMatchFoundScreenState extends State<RecordMatchFoundScreen>
    with TickerProviderStateMixin {
  double height = 0.0;
  double width = 0.0;

  var isSent = "0";
  ValidatePatientForAddIndividualResponse? userData;
  AddAuthorizedIndividualRequestModel? addAuthorizedIndividualRequestModel;
  dynamic argumentData;
  GlobalKey<ScaffoldState> drawerScaffoldKey = GlobalKey<ScaffoldState>();

  @override
  void initState() {
    argumentData = Get.arguments;
    userData = argumentData[0]['responseData'];
    addAuthorizedIndividualRequestModel = argumentData[1]['accessData'];
    super.initState();
  }

  InsertAuthorizedIndividualModel? insertAuthorizedIndividualModel;
  AuthenticationService authenticationService = new AuthenticationService();

  sendCredential() async {
    setState(() {
      isSent = "1";
    });

    await authenticationService.addAuthPatient(
        addAuthorizedIndividualRequestModel!.toMap(), widget.userMatchType);
    if (authenticationService.response_Status_Code_API_14 == 200) {
      setState(() {
        insertAuthorizedIndividualModel =
            authenticationService.insertAuthorizedIndividualModel;
        if (insertAuthorizedIndividualModel!.insertStatus!) {
          Get.to(
              AuthAccessGranted(
                  locationPhone:
                      insertAuthorizedIndividualModel!.locationPhone),
              arguments: userData);
          isSent = "2";
        }
      });
    } else {
      if (authenticationService.response_Status_Code_API_14 != 205) {
        setState(() {
          isSent = "2";
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    height = MediaQuery.of(context).size.height;
    width = MediaQuery.of(context).size.width;

    return IdleWrapper(
      child: SafeArea(
        child: GestureDetector(
          onTap: () {
            setState(() {
              ApiPath.isSettingIconPopupVisible = false;
              ApiPath.isSettingIconPopupOpen.value = false;
            });
          },
          child: Scaffold(
            appBar: !Responsive.isMobile(context)
                ? CommonAppBar(
                    context,
                    TabController(length: 6, vsync: this, initialIndex: 0),
                    setState,
                    null,
                    "")

                    preferredSize: Size(0, 0),
                    child: SizedBox(),
                  ),
            body: Responsive(
              mobile: mobileView(),
              tablet: webView(),
              desktop: webView(),
              k4Desktop: webView(),
            ),
          ),
        ),
      ),
    );
  }

  Widget mobileView() {
