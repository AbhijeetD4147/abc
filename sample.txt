class ForgotPasswordScreen extends StatefulWidget {
  var Guid;
  var locationPhone;
  ForgotPasswordScreen({super.key, this.Guid = "", this.locationPhone});

  @override
  State<ForgotPasswordScreen> createState() => ForgotPasswordScreenState();
}

class ForgotPasswordScreenState extends State<ForgotPasswordScreen>
    with TickerProviderStateMixin {
  double height = 0.0;
  double width = 0.0;
  final maxLines = 1;

  bool isSendEmailPopupVisible = false;

  TextEditingController usernameTextEditingController = TextEditingController();
  bool _validateUsername = false;

  TextEditingController captchaTextEditingController = TextEditingController();
  bool _validateCaptcha = false;
  bool _captchaMatch = false;
  var captchaCode = "";

  TextEditingController passwordTextEditingController = TextEditingController();
  TextEditingController confirmPasswordTextEditingController =
      TextEditingController();
  bool _validatePassword = false;
  bool _validateConfirmPassword = false;
  String? _errorPasswordText;
  String? _errorConfirmPasswordText;

  String? locationPhone;

  bool flow1 = true;
  bool flow2 = false;
  bool flow3 = false;
  int isResendLinkSecondTime = 0;

  bool passwordCharLong = false;
  bool passwordUpperCase = false;
  bool passwordLowerCase = false;
  bool passwordNumber = false;
  bool passwordSpecialChars = false;

  ForgotPasswordGUIDVerificationResponseModel?
      forgotPasswordGUIDVerificationResponseModel;
  ForgotUpdatePasswordResponseModel? forgotUpdatePasswordResponseModel;

  AnimationController? controller;
  bool loading = true;
  bool? captchaLoading;
  String? logo;
  ResetLinkSendResponse? resetLinkSendResponse;
  List<HomeDataModel> homeDataList = [];
  AuthenticationService authenticationService = new AuthenticationService();
  HomeService homeService = new HomeService();

  @override
  void initState() {
    super.initState();
    controller = AnimationController(
      vsync: this,
      duration: Duration(minutes: 5),
    );
    if (widget.Guid.isNotEmpty) {
      verifyGUID();
    } else {
      getCaptchaCode();
    }
  }

  Future<void> getCaptchaCode() async {
    setState(() {
      loading = false;
      captchaLoading = true;
    });

    setState(() {
      logo = GlobalParams.LOGO;
    });
    await authenticationService.getCaptcha();
    if (authenticationService.response_Status_Code_API_11 == 200) {
      setState(() {
        captchaCode = authenticationService.captchaCode!;
        captchaLoading = false;
      });
    } else {
      if (authenticationService.response_Status_Code_API_11 != 205) {
        setState(() {
          captchaLoading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  Future<void> verifyGUID() async {
    setState(() {
      loading = true;
    });
    ForgotUsernameGuidRequestModel forgotUsernameGuidRequestModel =
        ForgotUsernameGuidRequestModel();

    forgotUsernameGuidRequestModel.patientGUID = widget.Guid;
    await authenticationService
        .verifyGuidForResetPassword(forgotUsernameGuidRequestModel);

    if (authenticationService.response_Status_Code_API_10 == 200) {
      setState(() {
        forgotPasswordGUIDVerificationResponseModel =
            authenticationService.forgotPasswordGUIDVerificationResponseModel;
        if (forgotPasswordGUIDVerificationResponseModel!.status!
                .toLowerCase() ==
            "true") {
          flow1 = false;
          flow2 = false;
          flow3 = false;
        } else {
          flow1 = false;
          flow2 = false;
          flow3 = true;
        }
        locationPhone =
            forgotPasswordGUIDVerificationResponseModel?.locationPhone;
      });

      await homeService.getHomeData("Other");
      if (homeService.response_Status_Code_API == 200) {
        setState(() {
          homeDataList = homeService.homeDataModel!;
          logo = homeDataList[0].logo;
        });

        GlobalParams.LOGO = logo!;
        setState(() {
          loading = false;
        });
      } else {
        if (homeService.response_Status_Code_API != 205) {
          setState(() {
            loading = false;
          });
          showToast(
            context,
            message:
                'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
            type: ToastificationType.error,
          );
        }
      }
    } else {
      if (authenticationService.response_Status_Code_API_10 != 205) {
        setState(() {
          loading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  String get timerString {
    Duration duration = controller!.duration! * controller!.value;
    return (duration.inSeconds % 60).toString().length == 2
        ? '0${duration.inMinutes} : ${(duration.inSeconds % 60).toString()} minutes'
        : '0${duration.inMinutes} : 0${(duration.inSeconds % 60).toString()} minutes';
  }

  Future<void> sendMail() async {
    setState(() {
      isResendLinkSecondTime = isResendLinkSecondTime + 1;
      loading = true;
    });

    ResetPasswordLinkRequestModel resetPasswordLinkRequestModel =
        ResetPasswordLinkRequestModel();
    resetPasswordLinkRequestModel.userName =
        usernameTextEditingController.text.trim();
    resetPasswordLinkRequestModel.urlName = GlobalParams.PRACTICE_NAME;
    await authenticationService
        .resetLinkForForgotPassword(resetPasswordLinkRequestModel);

    if (authenticationService.response_Status_Code_API_6 == 200) {
      setState(() {
        resetLinkSendResponse =
            authenticationService.resetLinkForgotPasswordSendResponse;
        if (resetLinkSendResponse!.userId! > 0) {
          if (resetLinkSendResponse!.isAccountLocked!) {
            loading = false;
            Get.to(LoginScreen(
              callFrom: "Account Locked",
              accountLockedDateTime:
                  resetLinkSendResponse!.accountLockedDateTime!,
            ));
          } else {
            controller!.reverse(
                from: controller!.value == 0.0 ? 1.0 : controller!.value);
            showToast(
              context,
              message: 'Reset password email sent',
              type: ToastificationType.success,
            );
            loading = false;
            flow1 = false;
            flow2 = true;
            flow3 = false;
          }
        }
      });
    } else {
      if (authenticationService.response_Status_Code_API_6 != 205) {
        setState(() {
          loading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

    height = MediaQuery.of(context).size.height;
    width = MediaQuery.of(context).size.width;

    return SafeArea(
      child: Scaffold(

                        setState(() {
                          if (passwordTextEditingController.text
                              .trim()
                              .isNotEmpty) {
                            if (passwordTextEditingController.text
                                .trim()
                                .lengthValidation) {
                              passwordCharLong = true;
                            } else {
                              passwordCharLong = false;
                            }
                            if (passwordTextEditingController.text
                                .trim()
                                .containsNumbers) {
                              passwordNumber = true;
                            } else {
                              passwordNumber = false;
                            }
                            if (passwordTextEditingController.text
                                .trim()
                                .containsSpecialChars) {
                              passwordSpecialChars = true;
                            } else {
                              passwordSpecialChars = false;
                            }
                            if (passwordTextEditingController.text

  Future<void> updatePassword(String guid, String password) async {
    PasswordModel passwordModel = PasswordModel();
    passwordModel.guid = guid;
    passwordModel.password = password;

    await authenticationService.changePassword(passwordModel);
    if (authenticationService.response_Status_Code_API_24 == 200) {
      setState(() {
        forgotUpdatePasswordResponseModel =
            authenticationService.forgotUpdatePasswordResponseModel;
        if (forgotUpdatePasswordResponseModel!.status!.toLowerCase() ==
            "true") {
          Get.offAll(LoginScreen(callFrom: "Update Password"));
        } else {
          showCustomDialog(
            context,
            title: forgotUpdatePasswordResponseModel!.text,
            firstButtonText: "Ok",
            secondButtonText: "",
            iconData: Symbols.warning,
            firstButtonFunction: () {
              Get.back();
            },
            secondButtonFunction: () {},
          );
        }
      });
    } else {
      if (authenticationService.response_Status_Code_API_24 != 205) {
        setState(() {
          loading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }
