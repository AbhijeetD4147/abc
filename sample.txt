class ForgotUsernameScreen extends StatefulWidget {
  var Guid = "";
  var locationPhone;

  ForgotUsernameScreen({super.key, this.Guid = "", locationPhone});

  @override
  State<ForgotUsernameScreen> createState() => ForgotUsernameScreenState();
}

class ForgotUsernameScreenState extends State<ForgotUsernameScreen>
    with TickerProviderStateMixin {
  TextEditingController emailTextEditingController = TextEditingController();
  bool _validateEmail = false;
  String? emailErrorText;

  TextEditingController captchaTextEditingController = TextEditingController();
  bool _validateCaptcha = false;
  bool _captchaMatch = false;
  var captchaCode = "";

  bool flow1 = true;
  bool flow2 = false;
  bool flow3 = false;
  bool flow4 = false;

  UsernameGuidModel? usernameGuidModel;
  AnimationController? controller;
  ResetLinkSendResponse? resetLinkSendResponse;
  AuthenticationService authenticationService = new AuthenticationService();

  bool loading = false;
  bool? captchaLoading;
  String? logo;

  List<HomeDataModel> homeDataList = [];
  final FocusNode _focusNode1 = FocusNode();
  final FocusNode _focusNode2 = FocusNode();

  String? locationPhone;

  @override
  void dispose() {
    emailTextEditingController.dispose();
    captchaTextEditingController.dispose();
    _focusNode1.dispose();
    _focusNode2.dispose();
    super.dispose();
  }

  String get timerString {
    Duration duration = controller!.duration! * controller!.value;
    return (duration.inSeconds % 60).toString().length == 2
        ? '0${duration.inMinutes} : ${(duration.inSeconds % 60).toString()} minutes'
        : '0${duration.inMinutes} : 0${(duration.inSeconds % 60).toString()} minutes';
  }

  @override
  void initState() {
    super.initState();
    controller = AnimationController(
      vsync: this,
      duration: Duration(minutes: 5),
    );
    if (widget.Guid.isNotEmpty) {
      verifyGUID();
    } else {
      getCaptchaCode();
    }
  }

  Future<void> getCaptchaCode() async {
    setState(() {
      loading = false;
      captchaLoading = true;
    });

    setState(() {
      logo = GlobalParams.LOGO;
    });

    await authenticationService.getCaptcha();
    if (authenticationService.response_Status_Code_API_11 == 200) {
      setState(() {
        captchaCode = authenticationService.captchaCode!;
        captchaLoading = false;
      });
    } else {
      if (authenticationService.response_Status_Code_API_11 != 205) {
        setState(() {
          captchaLoading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  Future<void> verifyGUID() async {
    setState(() {
      loading = true;
    });
    ForgotUsernameGuidRequestModel forgotUsernameGuidRequestModel =
        ForgotUsernameGuidRequestModel();
    forgotUsernameGuidRequestModel.patientGUID = widget.Guid;

    await authenticationService
        .verifyGuidForResetUsername(forgotUsernameGuidRequestModel);
    if (authenticationService.response_Status_Code_API_8 == 200) {
      setState(() {
        usernameGuidModel = authenticationService.usernameGuidModel;
        if (usernameGuidModel!.status!.toLowerCase() == "true") {
          flow1 = false;
          flow2 = false;
          flow3 = false;
          flow4 = true;
        } else {
          flow1 = false;
          flow2 = false;
          flow3 = true;
          flow4 = false;
        }
        locationPhone = usernameGuidModel?.locationPhone;
      });
      HomeService homeService = HomeService();
      await homeService.getHomeData("Other");
      if (homeService.response_Status_Code_API == 200) {
        setState(() {
          homeDataList = homeService.homeDataModel!;
          logo = homeDataList[0].logo;
        });

        GlobalParams.LOGO = logo!;
        setState(() {
          loading = false;
        });
      } else {
        if (homeService.response_Status_Code_API != 205) {
          setState(() {
            loading = false;
          });
          showToast(
            context,
            message:
                'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
            type: ToastificationType.error,
          );
        }
      }
    } else {
      if (authenticationService.response_Status_Code_API_8 != 205) {
        setState(() {
          loading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  sendMail() async {
    setState(() {
      loading = true;
    });

    ResetLinkSendRequestModel resetLinkSendRequestModel =
        ResetLinkSendRequestModel();
    resetLinkSendRequestModel.userEmail =
        emailTextEditingController.text.trim();
    resetLinkSendRequestModel.urlName = GlobalParams.PRACTICE_NAME;

    await authenticationService
        .resetLinkForForgotUserName(resetLinkSendRequestModel);

    if (authenticationService.response_Status_Code_API_5 == 200) {
      setState(() {
        resetLinkSendResponse =
            authenticationService.resetLinkForgotUsernameSendResponse;
        if (resetLinkSendResponse!.userId! > 0) {
          controller!.reverse(
              from: controller!.value == 0.0 ? 1.0 : controller!.value);
          showToast(
            context,
            message: 'Reset username email sent',
            type: ToastificationType.success,
          );
          loading = false;
          flow1 = false;
          flow2 = true;
          flow3 = false;
          flow4 = false;
        }
      });
    } else {
      if (authenticationService.response_Status_Code_API_5 != 205) {
        setState(() {
          loading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return SafeArea(
      child: Scaffold(
        body: loading
            ? Center(
                child: CircularProgressIndicator(),
              )
            : Responsive(
                mobile: mobileView(),
                tablet: webView(),
                desktop: webView(),
                k4Desktop: webView(),
              ),
      ),
    );
  }

  Widget mobileView() {
    return Column(
      children: [
        Flexible(
          flex: 1,
          child: logoSection(),
        ),
        Expanded(
          flex: 2,
          child: Container(
            color: ThemeSelection.BGColor,
            child: SingleChildScrollView(child: usernameSection()),
          ),
        ),
      ],
    );
  }

  Widget webView() {
    return Row(
      children: [
        Expanded(
          flex: 1,
          child: logoSection(),
        ),
        Container(
          color: ThemeSelection.greyBorderColor,
          width: 1.0,
        ),
        Expanded(
          flex: 1,
          child: Container(
            color: ThemeSelection.BGColor,
            child: usernameSection(),
          ),
        )
      ],
    );
  }

  Widget usernameSection() {
    return Column(
      mainAxisAlignment: MainAxisAlignment.center,
      children: [
        flow1
            ? flowOne()
            : flow2
                ? flowTwo()
                : flow3
                    ? flowThree()
                    : flowFour(),
      ],
    );
  }

  Widget flowOne() {
    return Padding(
      padding: EdgeInsets.symmetric(horizontal: 12.w, vertical: 2.h),
      child: Column(
        mainAxisAlignment: MainAxisAlignment.center,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Center(
            child: Padding(
              padding: EdgeInsets.only(bottom: 1.h),
              child: Icon(
                const IconData(
                  0xE164,
                  fontFamily: 'metro',
                ),
                size: Responsive.isMobile(context) ? 65.sp : 20.sp,
                color: ThemeSelection.iconDefaultTertiaryColor,
              ),
            ),
          ),
          Center(
            child: Text(
              'Recover your Username',
              style: TextStyle(
                color: ThemeSelection.secondaryTextColor,
                fontFamily: 'SegoeUi',
                letterSpacing: 0.5,
                fontSize: Responsive.isMobile(context) ? 20.sp : 5.sp,
              ),
            ),
          ),
          SizedBox(
            height: 2.h,
          ),
          Text(
            'We can help you reset your Username.\nFollow the instruction below.',
            style: TextStyle(
              color: ThemeSelection.secondaryTextColor,
              fontFamily: 'SegoeUi',
              letterSpacing: 0.5,
              fontSize: Responsive.isMobile(context)
                  ? 12.sp
                  : Responsive.isTablet(context)
                      ? 4.sp
                      : 3.sp,
            ),
          ),
          SizedBox(
            height: 4.h,
          ),
          customTextFormField(
            context,
            hintText: "Your Email Address",
            readOnly: false,
            textEditingController: emailTextEditingController,
            focusNode: _focusNode1,
            maxLines: 1,
            onChanged: (val) {
              setState(() {
                if (emailTextEditingController.text.trim().isNotEmpty) {
                  if (emailTextEditingController.text.trim().emailValidate) {
                    _validateEmail = false;
                  } else {
                    _validateEmail = true;
                    emailErrorText = 'Please enter valid Email';
                  }
                } else {
                  _validateEmail = true;
                  emailErrorText = 'Email is required';
                }
              });
            },
            prefixIcon: Container(
              margin: EdgeInsets.only(right: 0.5.w),
              child: Icon(
                const IconData(0xE179, fontFamily: 'metro'),
                size: Responsive.isMobile(context) ? 12.sp : 3.sp,
                color: ThemeSelection.greyBorderColor,
              ),
            ),
            validate: _validateEmail,
            inlineError: false,
            key: ValueKey("Email"),
          ),
          _validateEmail
              ? Container(
                  decoration: BoxDecoration(
                    borderRadius: BorderRadius.circular(5),
                    color: ThemeSelection.textfieldInlineColor,
                  ),
                  alignment: Alignment.topLeft,
                  padding:
                      EdgeInsets.symmetric(vertical: 0.5.h, horizontal: 0.5.w),
                  child: Text(
                    emailErrorText!,
                    style: TextStyle(
                      color: Colors.white,
                    ),
                  ),
                )
              : SizedBox(),
          SizedBox(
            height: 2.h,
          ),
          Row(
            children: [
              Container(
                color: Colors.white,
                width: Responsive.isMobile(context) ? 20.w : 5.w,
                height: 5.h,
                child: Center(
                  child: Text(captchaCode,
                      style: TextStyle(
                        fontFamily: 'SegoeUi',
                        fontSize: Responsive.isMobile(context) ? 15.sp : 4.sp,
                        fontWeight: FontWeight.bold,
                        color: ThemeSelection.greyBorderColor,
                      ),
                      key: ValueKey("CaptchaCode")),
                ),
              ),
              SizedBox(
                width: 2.w,
              ),
              Padding(
                padding: EdgeInsets.symmetric(vertical: 1.h),
                child: InkWell(
                  onTap: () {
                    setState(() {
                      _captchaMatch = false;
                      captchaCode = "";
                    });
                    getCaptchaCode();
                  },
                  child: Text(
                    "Refresh",
                    style: TextStyle(
                      fontFamily: 'SegoeUi',
                      letterSpacing: 0.5,
                      decoration: TextDecoration.underline,
                      decorationColor: ThemeSelection.secondaryTextColor,
                      fontSize: Responsive.isMobile(context) ? 12.sp : 3.sp,
                      color: ThemeSelection.secondaryTextColor,
                    ),
                    key: ValueKey("Refresh Captcha"),
                  ),
                ),
              ),
            ],
          ),
          SizedBox(
            height: 1.h,
          ),
          Text(
            'Input symbols',
            style: TextStyle(
              color: ThemeSelection.secondaryTextColor,
              fontFamily: 'SegoeUi',
              letterSpacing: 0.5,
              fontSize: Responsive.isMobile(context) ? 15.sp : 4.sp,
            ),
          ),
          SizedBox(
            height: 2.h,
          ),
          customTextFormField(
            context,
            hintText: "",
            readOnly: false,
            textEditingController: captchaTextEditingController,
            focusNode: _focusNode2,
            maxLines: 1,
            onChanged: (val) {
              setState(() {
                if (captchaTextEditingController.text
                    .toString()
                    .trim()
                    .isNotEmpty) {
                  if (captchaTextEditingController.text.toString().trim() ==
                      captchaCode.trim()) {
                    _validateCaptcha = false;
                    _captchaMatch = false;
                  } else {
                    _validateCaptcha = false;
                    _captchaMatch = true;
                  }
                } else {
                  _captchaMatch = false;
                  _validateCaptcha = true;
                }
              });
            },
            validate: _validateCaptcha,
            inlineError: false,
            key: ValueKey("CaptchaCode Textfield"),
          ),
          _validateCaptcha || _captchaMatch
              ? Container(
                  decoration: BoxDecoration(
                    borderRadius: BorderRadius.circular(5),
                    color: ThemeSelection.textfieldInlineColor,
                  ),
                  alignment: Alignment.topLeft,
                  padding:
                      EdgeInsets.symmetric(vertical: 0.5.h, horizontal: 0.5.w),
                  child: Text(
                    _validateCaptcha
                        ? 'Captcha Code is required'
                        : _captchaMatch
                            ? 'Entered Wrong Captcha Code'
                            : "",
                    style: TextStyle(
                      color: Colors.white,
                    ),
                  ),
                )
              : SizedBox(),
          SizedBox(
            height: 3.h,
          ),
          Row(
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: <Widget>[
              Flexible(
                key: ValueKey("Cancel"),
                child: ghostButton(
                  context,
                  "Cancel",
                  () {
                    Get.back();
                  },
                ),
              ),
              SizedBox(
                width: Responsive.isMobile(context) ? 5.w : 1.w,
              ),
              Flexible(
                key: ValueKey("Next"),
                child: ghostButton(
                  context,
                  "Next",
                  () {
                    if (emailTextEditingController.text.trim().isNotEmpty) {
                      if (captchaTextEditingController.text.trim().isNotEmpty) {
                        if (captchaTextEditingController.text.trim() ==
                            captchaCode.trim()) {
                          sendMail();
                        }
                      }
                    }
                    if (emailTextEditingController.text.trim().isEmpty) {
                      setState(() {
                        _validateEmail = true;
                        emailErrorText = 'Email is required';
                      });
                    }
                    if (captchaTextEditingController.text.trim().isEmpty) {
                      setState(() {
                        _validateCaptcha = true;
                      });
                    }
                    if (captchaTextEditingController.text.trim().isNotEmpty) {
                      if (captchaTextEditingController.text.trim() !=
                          captchaCode.trim()) {
                        setState(() {
                          _captchaMatch = true;
                        });
                      }
                    }
                  },
                ),
              ),
            ],
          ),
        ],
      ),
    );
  }
