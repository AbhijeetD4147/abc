  //To Check OTP Entered By User
  Future<void> checkOtp(
      String userType, String authId, String otp, String callFrom) async {
    if (userType == 'New_Patient_Account') {
      customerData = {
        'PracticeName': GlobalParams.PRACTICE_NAME,
        'OwnerId': GlobalParams.PT_CUSTOMER_ID,
        'OwnerType': userType,
        'OTP': otp,
      };
    } else if (userType == "NEW_AUTH_INDIVIDUAL") {
      customerData = {
        'PracticeName': GlobalParams.PRACTICE_NAME,
        'OwnerId': authId,
        'OwnerType': userType,
        'OTP': otp,
      };
    } else {
      if (callFrom == "SignUpAsPatient") {
        customerData = {
          'PracticeName': GlobalParams.PRACTICE_NAME,
          'UserId': GlobalParams.USER_ID,
          'OwnerType': userType,
          'OwnerId': GlobalParams.PT_CUSTOMER_ID,
          'OTP': otp,
          'CallFrom': "SignUpAsPatient",
        };
      } else {
        customerData = {
          'PracticeName': GlobalParams.PRACTICE_NAME,
          'UserId': GlobalParams.USER_ID,
          'OwnerType': userType,
          'OwnerId': GlobalParams.PT_CUSTOMER_ID,
          'OTP': otp,
        };
      }
    }
    try {
      String queryString = Uri(queryParameters: customerData).query;
      String url = ApiPath.baseApi + 'api/PatientPortal/VerifyOTP';
      var requestUrl = url + '?' + queryString;
      Response response = await BaseWebService().getWebAPI(
        requestUrl: requestUrl,
        token: GlobalParams.TOKEN,
        practiceName: GlobalParams.PRACTICE_NAME,
      );

      if (response.statusCode == 200 && response.body == "SESSION INVALID") {
        response_Status_Code_API_1 = 205;
      } else if (response.statusCode == 200) {
        verifyOtpResponseModel =
            await VerifyOtpResponseModel.fromJson(jsonDecode(response.body));
        response_Status_Code_API_1 = response.statusCode;
      } else {
        maximum_Calling_API_1 = maximum_Calling_API_1 + 1;
        if (maximum_Calling_API_1 < ApiPath.MaxAPICalling) {
          if (response.statusCode == 401) {
            await AuthenticationService.generateToken();
            await checkOtp(userType, authId, otp, callFrom);
          } else {
            Future.delayed(Duration(seconds: 1), () async {
              await checkOtp(userType, authId, otp, callFrom);
            });
          }

          response_Status_Code_API_1 = response.statusCode;
        }

  static Future<void> generateToken() async {
    try {
      Response firstResponse = await get(
        Uri.parse(ApiPath.baseApi + 'GetWelcomeformParamForAuthenticate'),
        headers: {
          "AccountId": GlobalParams.PRACTICE_NAME,
          "SessionGuid": GlobalParams.SESSION_GUID
        },
      );
      if (firstResponse.statusCode == 200) {
        var vendorInterfaceData =
            VendorDataResponseModel.fromJson(jsonDecode(firstResponse.body));

        final customerData = jsonEncode({
          "vendorId": vendorInterfaceData.vendorId,
          "vendorPassword": vendorInterfaceData.vendorPassword,
          "accountId": vendorInterfaceData.accountId
        });

        Response response = await post(
          Uri.parse(ApiPath.baseApi + 'Authenticate'),
          body: customerData,
          headers: {
            "content-type": "application/json",
            "SessionGuid": GlobalParams.SESSION_GUID
          },
        );
        GlobalParams.TOKEN = response.body;
      } else {
        print("Request failed with status code : ${firstResponse.statusCode}");
      }
    } on SocketException {
      print("No internet Connection");
    } catch (e) {
      print('caught error : $e');
    }
  }
