  Future<void> signIn() async {
    setState(() {
      loading = true;
    });

    LoginModel loginModel = new LoginModel(
      userName: usernameTextEditingController.text,
      password: passwordTextEditingController.text,
      geoLocation: "",
      callFrom: "",
    );

    await authenticationService.login(loginModel.toMap());
    if (authenticationService.response_Status_Code_API_25 == 200) {
      setState(() {
        if (authenticationService.loginResponseModel!.userType!.isNotEmpty) {
          if (authenticationService.loginResponseModel!.isAccountLocked!) {
            loading = false;
            accountLockedDateTime = authenticationService
                .loginResponseModel!.accountLockedDateTime!;
            ApiPath.loginAttempt =
                authenticationService.loginResponseModel!.loginAttemptsleft!;
          } else {
            if (authenticationService.loginResponseModel!.loginStatus == 0) {
              loading = false;
              widget.callFrom = "";
              ApiPath.loginAttempt =
                  authenticationService.loginResponseModel!.loginAttemptsleft!;
            } else {
              var response = authenticationService.loginResponseModel;
              GlobalParams.USER_ID = response!.userId.toString();

              GlobalParams.USER_TYPE = response.userType!;
              GlobalParams.PT_CUSTOMER_ID = response.ptCustomerID.toString();

              if (response.sessionGuid!.isNotEmpty)
                GlobalParams.SESSION_GUID = response.sessionGuid!;

              if (response.userType == "Auth") {
                GlobalParams.PT_CUSTOMER_ID_For_Audit_Log =
                    response.ptCustomerID.toString();
                GlobalParams.isSettingIconMenuItemsAccess = false;
              } else {
                GlobalParams.PT_CUSTOMER_ID_For_Audit_Log = "0";
                GlobalParams.MAXIMEYES_PATIENT_NUMBER =
                    response.maximeyesPatientNumber!;
                GlobalParams.PATIENT_DOB = response.ptDOB!;
                GlobalParams.userMaximeyesPatientNumber =
                    response.maximeyesPatientNumber!;
                GlobalParams.userDOB = response.ptDOB!;
                if (response.userProfilePhotoBase64!.isNotEmpty)
                  GlobalParams.USER_PROFILE_PHOTO_BASE64 =
                      Base64Decoder().convert(response.userProfilePhotoBase64!);
              }

              ApiPath.isLogin = true;
              ApiPath.isGlobalStartTimer = true;
              GlobalParams.isSettingIconMenuItemsAccess = false;
              usernameTextEditingController.clear();
              passwordTextEditingController.clear();
              loading = false;

              if (response.isOptedOut!) {
                Get.off(OptInScreen(
                    optedBy: response.optedOutBy!,
                    optedOutDate: response.optedOutDate!));
              } else if (!response.iS_DOB_VERIFIED!) {
                Get.off(DobConfirmationScreen(optedPatient: false));
              } else {
                Get.off(
                  VerificationOtpScreen(
                      userType: response.userType,
                      locationPhone: response.locationPhone),
                  arguments: AuthorizedIndividualTermsResponseModel(
                      authId: response.ptCustomerID, authName: ""),
                );
              }
            }
          }
        } else {
          usernameTextEditingController.clear();
          passwordTextEditingController.clear();
          loading = false;
          isValidUser = false;
          widget.callFrom = "";
          ApiPath.loginAttempt = 3;
        }
        locationPhone =
            authenticationService.loginResponseModel!.locationPhone ?? '';
      });
    } else {
      if (authenticationService.response_Status_Code_API_25 != 205) {
        setState(() {
          loading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  void fieldValidation() {
    if (usernameTextEditingController.text.isEmpty ||
        passwordTextEditingController.text.isEmpty) {
      if (usernameTextEditingController.text.isEmpty) {
        setState(() {
          _validateUsername = true;
        });
      }
      if (passwordTextEditingController.text.isEmpty) {
        setState(() {
          _validatePassword = true;
        });
      }
    } else if (_formKey.currentState!.validate()) {
      signIn();
    }
  }

                        : 3.sp,
              ),
            ),
          ),
          Container(
            key: ValueKey("Back to Sign In"),
            child: Padding(
              padding: EdgeInsets.symmetric(vertical: 1.h),
              child: ghostButton(
                context,
                "Back to Sign In",
                () {
                  //Here add 5hrs and 30Mins to convert time into Indian Time
                  Duration difference = DateTime.now().difference(
                      DateTime.parse(accountLockedDateTime)
                          .add(Duration(hours: 5, minutes: 30)));
                  setState(() {
                    accountUnlockedDateTime =
                        DateTime.parse(accountLockedDateTime)
                            .add(Duration(hours: 5, minutes: 30))
                            .add(Duration(minutes: 15));
                  });

                  if (difference.inMinutes > 15) {
                    setState(() {
                      widget.callFrom = "";
                      ApiPath.loginAttempt = 3;
                      isValidUser = true;
                    });
                  } else {
                    showToast(
                      context,
                      message:
                          'Please retry after 15 mins from ${DateFormat.jm().format(DateTime.parse(accountLockedDateTime).toLocal().add(Duration(hours: 5, minutes: 30)))}.',
                      type: ToastificationType.warning,
                    );
                  }
                },
              ),
            ),
          ),
        ],
      ),
    );
  }
