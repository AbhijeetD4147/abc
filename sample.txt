  void validateManualDob(String authExpireDate) {
    // MM/DD/YYYY
    // 02/26/2023
    int month = int.parse(authExpireDate.substring(0, 2));
    int date = int.parse(authExpireDate.substring(3, 5));
    int year = int.parse(authExpireDate.substring(6, 10));

    DateTime checkDate = DateFormat("MM/dd/yyyy").parse(authExpireDate);
    DateTime today = DateTime.now();
    DateTime todayWithoutTime = DateTime(today.year, today.month, today.day);

    bool isNotLeapYear =
        ((year % 4 == 0) && (year % 100 != 0) || (year % 400 == 0));

    setState(() {
      if (month < 1 || month > 12) {
        _validateExpiryDate = true;
      } else if (date < 1 || date > 31) {
        _validateExpiryDate = true;
      } else if (year > 2300) {
        _validateExpiryDate = true;
      } else if (date > 29 && month == 2) {
        _validateExpiryDate = true;
      } else if (date == 31 &&
          (month == 2 ||
              month == 4 ||
              month == 6 ||
              month == 9 ||
              month == 11)) {
        _validateExpiryDate = true;
      } else if (checkDate.isBefore(todayWithoutTime)) {
        _validateExpiryDate = true;
      } else if (!isNotLeapYear && month == 2 && date == 29) {
        _validateExpiryDate = true;
      } else {
        _validateExpiryDate = false;
      }
      if (_validateExpiryDate) {
        dateFieldValidate = false;
      } else {
        dateFieldValidate = true;
      }
    });
  }

  String generateRandomChar() {
    final random = Random();
    // Generate a random number between 0 and 1
    final randomNumber = random.nextInt(2);
    // If the random number is 0, return an asterisk, otherwise, return a random digit
    return randomNumber == 0 ? '*' : random.nextInt(10).toString();
  }

  Widget userDetails() {
    final authDOBSpans = List<InlineSpan>.generate(
      authDob.length,
      (index) {
        final character = authDob[index];
        final shouldBeHidden = [1, 4, 8].contains(index);

        if (shouldBeHidden) {
          // Replace characters at the specified indices with asterisks
          return TextSpan(
            text: "*",
            style: CommonTextStyle.of(context)
                .H20_SUB(ThemeSelection.primaryTextColor),
          );
        } else {
          // Keep the rest of the characters as they are
          return TextSpan(
            text: character,
            style: CommonTextStyle.of(context)
                .H20_SUB(ThemeSelection.primaryTextColor),
          );
        }
      },
    );

    final mobileNumberWithoutHyphens = authMobile.replaceAll('-', '');
    final authMobileSpans = List<InlineSpan>.generate(
      mobileNumberWithoutHyphens.length,
      (index) {
        final character = mobileNumberWithoutHyphens[index];

        // Check if the current index is the 3rd, 6th, or 8th digit
        if (index == 2 || index == 5 || index == 7) {
          return TextSpan(
            text: '*',
            style: CommonTextStyle.of(context)
                .H20_SUB(ThemeSelection.primaryTextColor),
          );
        } else {
          return TextSpan(
            text: character,
            style: CommonTextStyle.of(context)
                .H20_SUB(ThemeSelection.primaryTextColor),
          );
        }
      },
    );

    final emailLength = authEmail.length;

    final seed =
        emailLength; // Use the email length as the seed for consistent character hiding

    final rng = Random(seed);

// Determine the number of characters to hide
    final minCharactersToHide = max(emailLength ~/ 2, 1);

    final hiddenIndices = Set<int>();

    while (hiddenIndices.length < minCharactersToHide) {
      final randomIndex = rng.nextInt(emailLength);
      if (authEmail[randomIndex] != '@' &&
          !authEmail
              .substring(
                  max(randomIndex - 3, 0), min(randomIndex + 4, emailLength))
              .contains(".com") &&
          !hiddenIndices.contains(randomIndex)) {
        hiddenIndices.add(randomIndex);
      }
    }

    final authEmailSpans = List<InlineSpan>.generate(
      emailLength,
      (index) {
        final character = authEmail[index];

        if (authEmail[index] == '@' ||
            authEmail
                .substring(max(index - 3, 0), min(index + 4, emailLength))
                .contains(".com")) {
          // Keep "@" and ".com" characters as they are
          return TextSpan(
            text: character,
            style: CommonTextStyle.of(context)
                .H20_SUB(ThemeSelection.primaryTextColor),
          );
        } else if (hiddenIndices.contains(index)) {
          // Replace characters at the chosen indices with asterisks
          return TextSpan(
            text: "*",
            style: CommonTextStyle.of(context)
                .H20_SUB(ThemeSelection.primaryTextColor),
          );
        } else {
          // Keep the rest of the characters as they are
          return TextSpan(
            text: character,
            style: CommonTextStyle.of(context)
                .H20_SUB(ThemeSelection.primaryTextColor),
          );
        }
      },
    );

    return Align(
      alignment: Responsive.isMobile(context)
          ? Alignment.centerLeft
          : Alignment.center,
      child: Column(
        mainAxisAlignment: MainAxisAlignment.start,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(
            'Authorized Individual',
            style: CommonTextStyle.of(context)
                .H18_SUB(ThemeSelection.primaryTextColor),
          ),
          SizedBox(
            height: Responsive.isMobile(context) ? 5.0 : height * 0.025,
          ),
          Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              Text(
                "Name: ",
                style: CommonTextStyle.of(context)
                    .H20_SUR(ThemeSelection.primaryTextColor),
              ),
              Text(
                authName,
                style: CommonTextStyle.of(context)
                    .H20_SUB(ThemeSelection.primaryTextColor),
                key: ValueKey("AuthName"),
              ),
            ],
          ),
          Visibility(
            visible: authDob.isBlank! ? false : true,
            child: SizedBox(
              height: Responsive.isMobile(context) ? 5.0 : width * 0.002,
            ),
          ),
          Visibility(
            visible: authDob.isBlank! ? false : true,
            child: Row(
              mainAxisSize: MainAxisSize.min,
              children: [
                Text(
                  "DOB: ",
                  style: CommonTextStyle.of(context)
                      .H20_SUR(ThemeSelection.primaryTextColor),
                ),
                RichText(
                  text: TextSpan(
                    children: authDOBSpans,
                  ),
                  key: ValueKey("DOB"),
                ),
              ],
            ),
          ),
          SizedBox(
            height: Responsive.isMobile(context) ? 5.0 : width * 0.002,
          ),
          Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              Text(
                "Phone: ",
                style: CommonTextStyle.of(context)
                    .H20_SUR(ThemeSelection.primaryTextColor),
              ),
              RichText(
                text: TextSpan(
                  children: authMobileSpans,
                ),
                key: ValueKey("Mobile"),
              ),
            ],
          ),
          SizedBox(
            height: Responsive.isMobile(context) ? 5.0 : width * 0.002,
          ),
          Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              Text(
                "Email: ",
                style: CommonTextStyle.of(context)

  Widget buildDateField() {
    return Container(
      key: ValueKey("Expiration Date"),
      alignment: Alignment.center,
      width: Responsive.isTablet(context) ? width * 0.15 : width * 0.1,
      child: customTextFormField(
        context,
        errorText: isDateChange ? "Invalid Expiry Date" : "",
        keyBoardType: TextInputType.phone,
        inputFormatters: [maskFormatterExpiryDate],
        hintText: "MM/DD/YYYY",
        inlineError: true,
        onChanged: (val) {
          setState(() {
            isExpired = false;
            isDateChange = true;
          });
          if (expiryDate.text.trim().isNotEmpty) {
            setState(() {
              _validateExpiryDate = true;
            });
            if (expiryDate.text.trim().length == 10) {
              validateManualDob(val);
            }
          } else {
            setState(() {
              _validateExpiryDate = false;
            });
          }
        },
        suffixIcon: InkWell(
          key: ValueKey("Date Picker"),
          onTap: () => _selectDate(context),
          child: Padding(
            padding: Responsive.isMobile(context)
                ? EdgeInsets.fromLTRB(10, 4, 7, 6)
                // .copyWith(top: 7)
                : Responsive.isTablet(context)
                    ? EdgeInsets.symmetric(horizontal: 11, vertical: 3)
                    : EdgeInsets.symmetric(horizontal: 4, vertical: 4),
            child: Icon(
              // const IconData(
              //   0xE101,
              //   fontFamily: 'metro',
              // ),
              Symbols.calendar_month,
              //size: Responsive.isMobile(context) ? 25.0 : height * 0.03,
              size: Responsive.isTablet(context)
                  ? height * 0.032
                  : Responsive.isMobile(context)
                      ? 25.0
                      : (MediaQuery.of(context).size.height >= 2560 &&
                              MediaQuery.of(context).size.width >= 1440)
                          ? height * 0.03 * 0.85
                          : height * 0.03,
              color: ThemeSelection.iconDefaultQuaternaryColor,
            ),
          ),
        ),
        textEditingController: expiryDate,
        validate: isExpired ? isExpired : _validateExpiryDate,
        validator: (value) {
          if (_validateExpiryDate) {
            return '';
          }
          return null;
        },
      ),
    );
  }

  void fieldValidation() {
    if (expiryDate.text.trim().isEmpty ||
        (expiryDate.text.trim().isNotEmpty &&
            dateFieldValidate &&
            !isExpired &&
            !_validateExpiryDate)) {
      showCustomDialog(
        context,
        title: "Do you want to update access permissions for the ${authName}?",
        firstButtonText: "Yes",
        secondButtonText: "No",
        iconData: Symbols.warning,
        firstButtonFunction: () {
          updateAuthPermission();
          Navigator.pop(context);
        },
        secondButtonFunction: () {
          Get.back();
          if (widget.isUserComingFromAuthPatientListScreen!) {
            Navigator.of(context).pop("");
          } else {
            Navigator.of(context).push(
              MaterialPageRoute(
                builder: (context) => AuthPatientList(),
              ),
            );
          }
        },
      );
    } else {
      setState(() {
        isExpired = false;
        _validateExpiryDate = true;
      });
      showToast(
        context,
        message: 'Expiration Date is Invalid',
        type: ToastificationType.warning,
      );
    }
  }
