  Future<void> patientValidate() async {
    setState(() {
      isButtonClicked = true;
    });

    patientInformation = new PatientInformation(
      firstName.text.trim(),
      lastName.text.trim(),
      countryCode.text.trim(),
      mobileNumber.text.trim(),
      email.text.trim(),
      dob.text.trim(),
    );

    Map<String, String> patientValidateData = {};

    if (widget.Title == "Sign Up as Patient") {
      patientValidateData = {
        'PracticeName': GlobalParams.PRACTICE_NAME,
        'FirstName': patientInformation!.firstName,
        'LastName': patientInformation!.lastName,
        'Mobile': patientInformation!.mobileNumber,
        'Email': patientInformation!.email,
        'DOB': patientInformation!.dob,
        'AuthSignUpAsPatient':
            widget.Title == "Sign Up as Patient" ? "true" : "false"
      };
    } else {
      patientValidateData = {
        'PracticeName': GlobalParams.PRACTICE_NAME,
        'FirstName': patientInformation!.firstName,
        'LastName': patientInformation!.lastName,
        'Mobile': countryCode.text + patientInformation!.mobileNumber,
        'Email': patientInformation!.email,
        'DOB': patientInformation!.dob,
        'AuthSignUpAsPatient': "false"
      };
    }

    AuthenticationService authenticationService = new AuthenticationService();

    await authenticationService.validatePatient(patientValidateData);
    if (authenticationService.response_Status_Code_API_7 == 200) {
      setState(() {
        patientAvailabilityResponseModel =
            authenticationService.patientAvailabilityResponseModel;
        if (patientAvailabilityResponseModel!.isExist!) {
          isButtonClicked = false;
          Get.off(
            PatientRecordMatchFoundScreen(
              email: patientInformation!.email,
              patientNumber: patientAvailabilityResponseModel!.patientNumber!,
            ),
          );
        } else {
          if (patientAvailabilityResponseModel!.accountType! ==
              "AuthSignUpAsPatient") {
            Get.offAll(SignUpCompleted());
          } else {
            isButtonClicked = false;
            Get.to(RecordNotMatchScreen());
          }
        }
      });
    } else {
      if (authenticationService.response_Status_Code_API_7 != 205) {
        setState(() {
          isButtonClicked = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  void fieldValidation() {
    if (firstName.text.trim().isNotEmpty &&
        lastName.text.trim().isNotEmpty &&
        countryCode.text.trim().isNotEmpty &&
        mobileNumber.text.trim().isNotEmpty &&
        email.text.trim().isNotEmpty &&
        dob.text.trim().isNotEmpty &&
        mobileNumber.text.mobileValidate &&
        mobileNumber.text != '000-000-0000' &&
        email.text.emailValidate &&
        dateFieldValidate) {
      patientValidate();
    } else {
      setState(() {
        if (firstName.text.trim().isNotEmpty) {
          _validateFirstName = false;
        } else {
          _validateFirstName = true;
        }
        if (lastName.text.trim().isNotEmpty) {
          _validateLastName = false;
        } else {
          _validateLastName = true;
        }
        if (mobileNumber.text.trim().isNotEmpty) {
          if (mobileNumber.text.mobileValidate &&
              mobileNumber.text.trim() != '000-000-0000') {
            _validateMobileNumber = false;
          } else {
            mobileNumberErrorText = 'Please enter valid mobile number';
            _validateMobileNumber = true;
          }
        } else {
          _validateMobileNumber = true;
          mobileNumberErrorText = "Mobile is required";
        }
        if (email.text.trim().isNotEmpty) {
          if (email.text.trim().emailValidate) {
            _validateEmail = false;
          } else {
            _validateEmail = true;
            emailErrorText = 'Please enter valid Email';
          }
        } else {
          _validateEmail = true;
          emailErrorText = 'Email is required';
        }
        if (dob.text.trim().isNotEmpty) {
          _validateDob = false;
          if (dob.text.trim().length == 10) {
            validateManualDob(dob.text.trim());
          } else {
            _validateManualDob = true;
          }
        } else {
          _validateDob = true;
        }
      });
    }
  }

  void validateManualDob(String dateOfBirth) {
    // MM/DD/YYYY
    // 02/26/2023
    int month = int.parse(dateOfBirth.substring(0, 2));
    int date = int.parse(dateOfBirth.substring(3, 5));
    int year = int.parse(dateOfBirth.substring(6, 10));

    DateTime checkDate = DateFormat("MM/dd/yyyy").parse(dateOfBirth);
    DateTime today = DateTime.now();

    bool isNotLeapYear =
        ((year % 4 == 0) && (year % 100 != 0) || (year % 400 == 0));

    setState(() {
      if (month < 1 || month > 12) {
        _validateManualDob = true;
      } else if (date < 1 || date > 31) {
        _validateManualDob = true;
      } else if (year < 1900) {
        _validateManualDob = true;
      } else if (date > 29 && month == 2) {
        _validateManualDob = true;
      } else if (date == 31 &&
          (month == 2 ||
              month == 4 ||
              month == 6 ||
              month == 9 ||
              month == 11)) {
        _validateManualDob = true;
      } else if (checkDate.compareTo(today) > 0) {
        _validateManualDob = true;
      } else if (!isNotLeapYear && month == 2 && date == 29) {
        _validateManualDob = true;
      } else {
        _validateManualDob = false;
      }
      if (_validateManualDob) {
        dateFieldValidate = false;
      } else {
        dateFieldValidate = true;
      }
    });
  }

  Future _selectDate(BuildContext context) async {
    final DateTime? pickedDate = await showDatePicker(

        initialDate: DateTime.now(),
        firstDate: DateTime(1900),
        lastDate: DateTime.now(),
        initialEntryMode: DatePickerEntryMode.calendarOnly,
        cancelText: "CLOSE",
        confirmText: "SAVE",
        builder: (context, child) {
          return Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Theme(
                data: Theme.of(context).copyWith(
                  colorScheme: ColorScheme.light(
                    primary: ThemeSelection.BGColor, // header background color
                    onPrimary: Colors.white, // header text color
                    onSurface: Colors.black, // body text color
                  ),
                  textButtonTheme: TextButtonThemeData(
                    style: ButtonStyle(
                      foregroundColor: WidgetStateProperty.resolveWith<Color>(
                          (Set<WidgetState> states) {
                        if (states.contains(WidgetState.hovered))
                          return ThemeSelection
                              .primaryPressedBackgroundButtonColor;
                        if (states.contains(WidgetState.focused))
                          return ThemeSelection.whiteButtonHover;
                        return ThemeSelection.BGColor;
                      }),
                    ),
                  ),
                ),
                child: ConstrainedBox(
                  constraints: BoxConstraints(
                    minWidth: Responsive.isMobile(context)
                        ? 100.w
                        : Responsive.isTablet(context)
                            ? 45.w
                            : 23.w,
                    minHeight: Responsive.isMobile(context)
                        ? 70.h
                        : Responsive.isTablet(context)
                            ? 70.h
                            : 60.h,
                  ),
                  child: child,
                ),
              ),
            ],
          );
        });
    if (pickedDate != null) {
      setState(() {
        dob.text = dateFormat.format(pickedDate);
        _validateDob = false;
        _validateManualDob = false;
        dateFieldValidate = true;
      });
    }
  }
