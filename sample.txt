            children: [

                  colorScheme: ColorScheme.light(
                    primary: ThemeSelection.BGColor, // header background color
                    onPrimary: Colors.white, // header text color
                    onSurface: Colors.black, // body text color
                  ),

                    style: ButtonStyle(

                    ),
                  ),
                ),

                  constraints: BoxConstraints(
                    minWidth: Responsive.isMobile(context)
                        ? 100.w

                    minHeight: Responsive.isMobile(context)
                        ? 70.h

      setState(() {
        expireDate.text = dateFormat.format(pickedDate);
        dateFieldValidate = true;
        _validateExpiryDate = false;
      });
    }
  }

  void validateManualDob(String authExpireDate) {
    // MM/DD/YYYY
    // 02/26/2023
    int month = int.parse(authExpireDate.substring(0, 2));
    int date = int.parse(authExpireDate.substring(3, 5));
    int year = int.parse(authExpireDate.substring(6, 10));

    DateTime checkDate = DateFormat("MM/dd/yyyy").parse(authExpireDate);
    DateTime today = DateTime.now();
    DateTime todayWithoutTime = DateTime(today.year, today.month, today.day);

    bool isNotLeapYear =
        ((year % 4 == 0) && (year % 100 != 0) || (year % 400 == 0));

    setState(() {
      if (month < 1 || month > 12) {
        _validateExpiryDate = true;
      } else if (date < 1 || date > 31) {
        _validateExpiryDate = true;
      } else if (year > 2300) {
        _validateExpiryDate = true;
      } else if (date > 29 && month == 2) {
        _validateExpiryDate = true;
      } else if (date == 31 &&
          (month == 2 ||
              month == 4 ||
              month == 6 ||
              month == 9 ||
              month == 11)) {
        _validateExpiryDate = true;
      } else if (checkDate.isBefore(todayWithoutTime)) {
        _validateExpiryDate = true;
      } else if (!isNotLeapYear && month == 2 && date == 29) {
        _validateExpiryDate = true;
      } else {
        _validateExpiryDate = false;
      }
      if (_validateExpiryDate) {
        dateFieldValidate = false;
      } else {
        dateFieldValidate = true;
      }
    });
  }

  void fieldValidations() {
    if (firstName.text.trim().isNotEmpty &&
        lastName.text.trim().isNotEmpty &&
        countryCode.text.isNotEmpty &&
        mobileNumber.text.trim().isNotEmpty &&
        email.text.trim().isNotEmpty &&
        mobileNumber.text.mobileValidate &&
        mobileNumber.text != '000-000-0000' &&
        email.text.emailValidate &&
        (expireDate.text.trim().isEmpty ||
            (expireDate.text.trim().isNotEmpty && dateFieldValidate))) {
      userCheckApi();
    } else {
      setState(() {
        if (firstName.text.trim().isNotEmpty) {
          _validateFirstName = false;
        } else {
          _validateFirstName = true;
        }
        if (lastName.text.trim().isNotEmpty) {
          _validateLastName = false;
        } else {
          _validateLastName = true;
        }
        if (mobileNumber.text.trim().isNotEmpty) {
          if (mobileNumber.text.mobileValidate &&
              mobileNumber.text.trim() != '000-000-0000') {
            _validateMobileNumber = false;
          } else {
            mobileNumberErrorText = 'Please enter valid mobile number';
            _validateMobileNumber = true;
          }
        } else {
          _validateMobileNumber = true;
          mobileNumberErrorText = "Mobile is required";
        }
        if (email.text.trim().isNotEmpty) {
          if (email.text.trim().emailValidate) {
            _validateEmail = false;
          } else {
            _validateEmail = true;
            emailErrorText = 'Please enter valid Email';
          }
        } else {
          _validateEmail = true;
          emailErrorText = 'Email is required';
        }
        if (expireDate.text.trim().isNotEmpty) {
          if (dateFieldValidate) {
            _validateExpiryDate = false;
          } else {
            _validateExpiryDate = true;
          }
        }
      });
    }
  }

  Future<void> userCheckApi() async {
    setState(() {
      isButtonClicked = true;
    });

    Map<String, String> authPatientValidateData = {
      'PracticeName': GlobalParams.PRACTICE_NAME,
      'FirstName': firstName.text.trim(),
      'LastName': lastName.text.trim(),
      'Mobile': mobileNumber.text.isNotEmpty ? mobileNumber.text.trim() : "",
      'Email': email.text.isNotEmpty ? email.text.trim() : "",
      'UserId': GlobalParams.USER_ID
    };

    await authenticationAuthUserService
        .validateAuthPatient(authPatientValidateData);
    if (authenticationAuthUserService.response_Status_Code_API_1 == 200) {
      setState(() async {
        validatePatientForAddIndividualResponse = authenticationAuthUserService
            .validatePatientForAddIndividualResponse;
        AddAuthorizedIndividualRequestModel addIndividualRequest =
            new AddAuthorizedIndividualRequestModel(
          firstName: firstName.text.trim(),
          lastName: lastName.text.trim(),
          mobile: mobileNumber.text.trim(),
          email: email.text.trim(),
          locationId: 0,
          userId: int.parse(GlobalParams.USER_ID),
          ptCustomerId:
              validatePatientForAddIndividualResponse!.ptCustomerId == null
                  ? 0
                  : validatePatientForAddIndividualResponse!.ptCustomerId,
          expiryDate: expireDate.text.isNotEmpty ? expireDate.text.trim() : "",
          reason: reason.text.isNotEmpty ? reason.text.trim() : "",
        );
        if (validatePatientForAddIndividualResponse!.isExist!) {
          setState(() {
            isButtonClicked = false;
          });
          if (GlobalParams.MAXIMEYES_PATIENT_NUMBER ==
              validatePatientForAddIndividualResponse!.patientNumber) {
            showToast(
              context,
              message: 'You cannot add yourself as Authorized User!',
              type: ToastificationType.warning,
            );
          } else if (validatePatientForAddIndividualResponse!.matchType ==
              "AuthAlreadyAdded") {
            showToast(
              context,
              message: 'Authorized User already exists!',
              type: ToastificationType.warning,
            );
          } else if (validatePatientForAddIndividualResponse!.matchType ==
              "AccountExist") {
            //Authorized User Added in Past
            showCustomDialog(
              context,
              title:
                  "Account already exists for ${validatePatientForAddIndividualResponse!.patientName}. Do you want to activate the same account again?",
              firstButtonText: "Activate Account",
              secondButtonText: "Cancel",
              iconData: Symbols.warning,
              firstButtonFunction: () async {
                await authenticationService.addAuthPatient(
                    addIndividualRequest.toMap(),
                    validatePatientForAddIndividualResponse!.matchType!);
                if (authenticationService.response_Status_Code_API_14 == 200) {
                  insertAuthIndividualModel =
                      authenticationService.insertAuthorizedIndividualModel;
                  if (insertAuthIndividualModel!.insertStatus!) {
                    Navigator.of(context).pop();
                    Navigator.of(context).push(
                      MaterialPageRoute(
                        builder: (context) => AuthManagePermission(
                          authId: insertAuthIndividualModel!.authIndividualId
                              .toString(),
                          isUserComingFromAuthPatientListScreen: false,
                        ),
                      ),
                    );
                  } else {
                    setState(() {
                      isButtonClicked = false;
                    });
                    ScaffoldMessenger.of(context).showSnackBar(
                      const SnackBar(
                        content: Text(
                            'Something went wrong! Please try again later.'),
                      ),
                    );
                  }
                } else {
                  setState(() {
                    isButtonClicked = false;
                  });
                  showToast(
                    context,
                    message:
                        'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
                    type: ToastificationType.error,
                  );
                }
              },
              secondButtonFunction: () {
                Get.back();
              },
            );
          } else if (validatePatientForAddIndividualResponse!
                      .patientNumber.isBlank! &&
                  validatePatientForAddIndividualResponse!.matchType ==
                      "ExistingAuthPartialMatch" ||
              validatePatientForAddIndividualResponse!.matchType ==
                  "ExistingPatientPartialMatch") {
            //Re-Enter Correct Details
            Get.to(
              RecordMatchFoundScreen(
                alertMsg:
                    "We have found a similar matching record\nwith us but with different details. Did you\nenter details correctly?",
                userMatchType:
                    validatePatientForAddIndividualResponse!.matchType!,
              ),
              arguments: [
                {'responseData': validatePatientForAddIndividualResponse},
                {'accessData': addIndividualRequest}
              ],
            );
          } else {
            //Grant Access Screen
            Get.to(
              RecordMatchFoundScreen(
                alertMsg: "We have found a matching record with us.",
                userMatchType:
                    validatePatientForAddIndividualResponse!.matchType!,
              ),
              arguments: [
                {'responseData': validatePatientForAddIndividualResponse},
                {'accessData': addIndividualRequest}
              ],
            );
          }
        } else {
          // Add New Patient
          await authenticationService.addAuthPatient(
              addIndividualRequest.toMap(),
              validatePatientForAddIndividualResponse!.matchType!);
          if (authenticationService.response_Status_Code_API_14 == 200) {
            insertAuthIndividualModel =
                authenticationService.insertAuthorizedIndividualModel;
            if (insertAuthIndividualModel!.insertStatus!) {
              Get.to(
                  AuthAccessGranted(
                      locationPhone: insertAuthIndividualModel!.locationPhone),
                  arguments: validatePatientForAddIndividualResponse);
              setState(() {
                isButtonClicked = false;
              });
            } else {
              setState(() {
                isButtonClicked = false;
              });
              showToast(
                context,
                message:
                    'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
                type: ToastificationType.error,
              );
            }
          } else {
            setState(() {
              isButtonClicked = false;
            });
            showToast(
              context,
              message:
                  'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
              type: ToastificationType.error,
            );
          }
        }
      });
    } else {
      if (authenticationAuthUserService.response_Status_Code_API_1 != 205) {
        setState(() {
          isButtonClicked = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }
