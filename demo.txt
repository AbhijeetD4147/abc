import '../../utils/idle_wrapper.dart';

class PatientRecordMatchFoundScreen extends StatefulWidget {
  final String email;
  final int patientNumber;

  PatientRecordMatchFoundScreen({
    super.key,
    required this.email,
    required this.patientNumber,
  });

  @override
  State<PatientRecordMatchFoundScreen> createState() =>
      _PatientRecordMatchFoundScreenState();
}

class _PatientRecordMatchFoundScreenState
    extends State<PatientRecordMatchFoundScreen> with TickerProviderStateMixin {
  var isSent = "0";
  dynamic argumentData;
  GlobalKey<ScaffoldState> drawerScaffoldKey = GlobalKey<ScaffoldState>();
  double height = 0.0;
  double width = 0.0;

  @override
  void initState() {
    argumentData = Get.arguments;
    super.initState();
  }

  AuthenticationService authenticationService = new AuthenticationService();

  sendCredential() async {
    setState(() {
      isSent = "1";
    });
    await authenticationService.sendCredentials(
        widget.email, widget.patientNumber);

    if (authenticationService.response_Status_Code_API_3 == 200) {
      if (authenticationService.isSent == "true") {
        Get.off(CredentialSendForExistUser(
            email: widget.email, patientNumber: widget.patientNumber));
      }
      setState(() {
        isSent = "2";
      });
    } else {
      if (authenticationService.response_Status_Code_API_3 != 205) {
        setState(() {
          isSent = "2";
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    height = MediaQuery.of(context).size.height;
    width = MediaQuery.of(context).size.width;
    return IdleWrapper(
      child: SafeArea(
        child: GestureDetector(
          onTap: () {
            setState(() {
              ApiPath.isSettingIconPopupVisible = false;
              ApiPath.isSettingIconPopupOpen.value = false;
            });
          },
          child: Scaffold(
            appBar: PreferredSize(
              preferredSize: Size(0, 0),
              child: SizedBox(),
            ),
            body: Responsive(
              mobile: mobileView(),
              tablet: webView(),
              desktop: webView(),
              k4Desktop: webView(),
            ),
          ),
        ),
      ),
    );
  }

  Widget mobileView() {
    return Column(
      children: [
        Expanded(
          flex: 2,
          child: Container(
            padding: EdgeInsets.symmetric(horizontal: 15.0),
            child: Center(child: recordMatchFoundSection()),
          ),
        ),
        Expanded(
          flex: 1,
          child: Container(
            width: width,
            color: ThemeSelection.textfieldFilledColor,
            child: sendCredentialSection(),
          ),
        )
      ],
    );
  }

  Widget webView() {
    return Stack(
      children: [
        Row(
          children: [
            Expanded(
              flex: 1,
              child: Container(
                color: ThemeSelection.textfieldFilledColor,
                child: sendCredentialSection(),
              ),
            ),
            Container(
              color: ThemeSelection.greyBorderColor,
              width: 1.0,
            ),
            Expanded(
              flex: 2,
              child: recordMatchFoundSection(),
            )
          ],
        ),
      ],
    );
  }

  Widget sendCredentialSection() {
    return Column(
      mainAxisAlignment: MainAxisAlignment.center,
      children: [
        Text(
          'Did you forget your login credentials?',
          style: CommonTextStyle.of(context)
              .H21_SUR(ThemeSelection.primaryTextColor),
        ),
        SizedBox(
          height: 30.0,
        ),
        isSent == "1"
            ? CircularProgressIndicator()
            : OutlinedButton(
                key: ValueKey("Send Credentials"),
                onPressed: () {
                  sendCredential();
                },
                child: Container(
                  padding: EdgeInsets.only(
                      left: 40.0, right: 40.0, top: 10.0, bottom: 10.0),
                  child: Text(
                    'Send me credentials',
                    style: TextStyle(
                        fontFamily: 'SegoeUi',
                        fontSize: 16.0,
                        fontWeight: FontWeight.normal,
                        color: ThemeSelection.primaryTextColor),
                  ),
                ),
                style: ButtonStyle(
                  shape: WidgetStateProperty.all<RoundedRectangleBorder>(
                    RoundedRectangleBorder(
                      borderRadius: BorderRadius.circular(5.0),
                      side: BorderSide(
                        color: ThemeSelection.primaryTextColor,
                        width: 1,
                      ),
                    ),
                  ),
                ),
              )
      ],
    );
  }

  Widget recordMatchFoundSection() {
    return Column(
      mainAxisSize: MainAxisSize.min,
      children: [
        Icon(
          Symbols.check_circle,
          size: 100.0,
        ),
        SizedBox(
          height: 20.0,
        ),
        Text(
          "Record match found",
          style: CommonTextStyle.of(context)
              .H15_SUR(ThemeSelection.primaryTextColor),
        ),
        SizedBox(
          height: Responsive.isMobile(context) ? 10.0 : 20.0,
        ),
        Text(
          "We have found a matching record with us.",
          style: CommonTextStyle.of(context)
              .H21_SUR(ThemeSelection.primaryTextColor),
        ),
        SizedBox(
          height: 5.0,
        ),
        Text(
          "Please Sign In using your password",
          style: CommonTextStyle.of(context)
              .H20_SUR(ThemeSelection.primaryTextColor),
        ),
        SizedBox(
          height: 60.0,
        ),
        Container(
          key: ValueKey("Sign In"),
          child: primaryButton(
            context,
            () {
              Get.offAll(LoginScreen());
            },
            "Sign In",
            ThemeSelection.secondaryTextColor,
          ),
        ),
      ],
    );
  }
}
