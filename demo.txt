  String removeFontSizeAttribute(String html) {
    final fontSizePattern = RegExp(r'font-size:[^;]+;');
    return html.replaceAll(fontSizePattern, '');
  }

  downloadFile(String? base64File, String? fileName) {
    final blob = html.Blob([base64Decode(base64File!)]);
    final url = html.Url.createObjectUrlFromBlob(blob);
    final anchor = html.document.createElement('a') as html.AnchorElement
      ..href = url
      ..style.display = 'none'
      ..download = fileName;
    html.document.body!.children.add(anchor);
    anchor.click();
    html.document.body!.children.remove(anchor);
    html.Url.revokeObjectUrl(url);
  }

  String formatHtmlString(String htmlString) {
    final liTagRegExp = RegExp(r'(<li[^>]*>\s*<a[^>]*>.*?<\/a>\s*<\/li>)',
        multiLine: true, caseSensitive: false);

    // Find all matches
    final matches =
        liTagRegExp.allMatches(htmlString).map((m) => m.group(0)).toList();

    if (matches.isEmpty) {
      return htmlString; // No list items found
    }

    // Number of columns (fixed to 3 for this example)
    var numColumns = Responsive.isMobile(context) ? 2 : 3;

    // Calculate items per segment
    int totalItems = matches.length;
    int itemsPerSegment = (totalItems / numColumns).ceil();

    List<String> segments = [];

    for (int i = 0; i < numColumns; i++) {
      // Get slice of items for this segment
      int start = i * itemsPerSegment;
      int end = (start + itemsPerSegment > totalItems)
          ? totalItems
          : start + itemsPerSegment;
      List<String?> itemsForThisSegment = matches.sublist(start, end);

      // Create the <ul> element for this segment
      String ulSegment = '<ul>\n' + itemsForThisSegment.join('\n') + '\n</ul>';
      segments.add('<td>$ulSegment</td>');
    }

    // Create a table to hold the columns
    String table =
        '<table>\n<tr>\n' + segments.join('\n') + '\n</tr>\n</table>';

    // Replace the original list items with a placeholder to avoid repetition
    String modifiedHtml = htmlString.replaceAll(liTagRegExp, '');

    // Find a place to insert the wrapped segments (e.g., after "Table of Contents" if it exists)
    final tocPosition = modifiedHtml.indexOf('Table of Contents');
    if (tocPosition != -1) {
      final insertPosition = modifiedHtml.indexOf('</ul>', tocPosition);
      if (insertPosition != -1) {
        modifiedHtml = modifiedHtml.replaceRange(
            insertPosition + 5, insertPosition + 5, table);
      } else {
        modifiedHtml =
            modifiedHtml.replaceRange(tocPosition, tocPosition, table);
      }
    } else {
      modifiedHtml = table + modifiedHtml;
    }

    modifiedHtml = addSpaceBetweenRows(modifiedHtml);

    // Add padding around the final HTML
    String paddedHtml = !Responsive.isMobile(context)
        ? modifiedHtml
        : '<div style="padding: 40px;">$modifiedHtml</div>';

    return paddedHtml;
  }

  String addSpaceBetweenRows(String htmlString) {
    // Parse the HTML string
    dom.Document document = parse(htmlString);
    dom.Element table = document.getElementsByTagName('table')[1];
    List<dom.Element> trs = table.getElementsByTagName('tr');

    // Add space between the phone, fax, and email rows
    for (int i = 0; i < trs.length; i++) {
      if (trs[i].text.contains('Home Phone:') ||
          trs[i].text.contains('Fax:') ||
          trs[i].text.contains('Cell Phone:') ||
          trs[i].text.contains('Day Phone:')) {
        // Create a new empty row
        dom.Element emptyRow = dom.Element.tag('tr');
        dom.Element emptyCell = dom.Element.tag('td');
        emptyCell.attributes['colspan'] = '10';
        emptyCell.text = ' ';
        emptyRow.append(emptyCell);

        // Insert the empty row after the current row
        table.insertBefore(emptyRow, trs[i].nextElementSibling);

        dom.Element emptyRow2 = dom.Element.tag('tr');
        dom.Element emptyCell2 = dom.Element.tag('td');
        emptyCell2.attributes['colspan'] = '10';
        emptyCell2.text = ' ';
        emptyRow.append(emptyCell2);

        // Insert the empty row after the current row
        table.insertBefore(emptyRow2, trs[i].nextElementSibling);

        dom.Element emptyRow3 = dom.Element.tag('tr');
        dom.Element emptyCell3 = dom.Element.tag('td');
        emptyCell3.attributes['colspan'] = '10';
        emptyCell3.text = ' ';
        emptyRow3.append(emptyCell3);

        // Insert the empty row after the current row
        table.insertBefore(emptyRow3, trs[i].nextElementSibling);
      }
    }

    // Convert the document back to an HTML string
    return document.outerHtml;
  }

  Future<void> sendMessage() async {
    if (subjectController.text.isNotEmpty &&
        toController.text.isNotEmpty &&
        !validateSubject &&
        !validateToEmail) {
      setState(() {
        healthSummaryForShareLoading = true;
      });

      DateFormat dateFormat = DateFormat("yyyy-MM-dd hh:mm");

      VisitSummaryTransmitModel visitSummaryTransmitModel =
          VisitSummaryTransmitModel();
      visitSummaryTransmitModel.ptcustomerId =
          int.parse(GlobalParams.PT_CUSTOMER_ID);
      visitSummaryTransmitModel.isActive = true;
      visitSummaryTransmitModel.toEmail = toController.text.trim();
      visitSummaryTransmitModel.messageSubject = subjectController.text.trim();
      visitSummaryTransmitModel.messageText = messageController.text.trim();
      visitSummaryTransmitModel.messageDateTime =
          dateFormat.format(currentDate);
      visitSummaryTransmitModel.messageStatus = "Sent";
      visitSummaryTransmitModel.isDirectMessage = false;
      visitSummaryTransmitModel.kno2Id = "";
      visitSummaryTransmitModel.isCustomer = true;
      visitSummaryTransmitModel.isIncoming = false;
      visitSummaryTransmitModel.isFavourite = true;
      visitSummaryAttachmentList.removeAt(1);
      visitSummaryTransmitModel.visitSummaryAttachmentList =
          visitSummaryAttachmentList;
      visitSummaryTransmitModel.isReplied = false;
      visitSummaryTransmitModel.parentMessageId = 0;
      visitSummaryTransmitModel.ptPracticePersonId = 118;
      visitSummaryTransmitModel.userId = int.parse(GlobalParams.USER_ID);
      visitSummaryTransmitModel.switchUserId =
          int.parse(GlobalParams.SWITCH_USER_ID);

      await healthSummaryService
          .transmitHealthSummary(visitSummaryTransmitModel);
      if (healthSummaryService.response_Status_Code_API_3 == 200) {
        setState(() {
          emailIdRequiredInPayloadForAuditLog = true;
          isVisitSummaryForwardScreen = false;
          isVisitSummaryClicked = false;
          isMessageSentScreenVisible = true;
          Future.delayed(Duration(seconds: 2), () {
            setState(() {
              isMessageSentScreenVisible = false;
            });
          });
          healthSummaryForShareLoading = false;
          messageController.clear();
          subjectController.clear();
        });
        await updateActivityForVisitSummary("Transmit CCDA");
      } else {
        if (healthSummaryService.response_Status_Code_API_3 != 205) {
          setState(() {
            // healthSummaryListLoading = false;
          });
          showToast(
            context,
            message:
                'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
            type: ToastificationType.error,
          );
        }
      }
    }
    if (subjectController.text.isEmpty) {
      setState(() {
        validateSubject = true;
      });
    }
    if (toController.text.isEmpty) {
      setState(() {
        validateToEmail = true;
        invalidEmailMessage = 'Email is required';
      });
    }
  }
