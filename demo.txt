  Future<void> deleteSelectedMessages(bool isDeleteingLatestThread) async {
    MessageService messageService = MessageService();
    selectedMessageIdsForAction!.length != 0
        ? await messageService.deleteMessage(
            selectedMessageIdsForAction!.join(","),
            searchTextController.text.trim().isNotEmpty
                ? tempMessageType == "Archive"
                    ? true
                    : false
                : showMessagesOfArchive)
        : isDeleteingLatestThread
            ? await messageService.deleteMessage(
                latestMessageThreadId.toString(),
                searchTextController.text.trim().isNotEmpty
                    ? tempMessageType == "Archive"
                        ? true
                        : false
                    : showMessagesOfArchive)
            : await messageService.deleteMessage(
                parentMessageId.toString(),
                searchTextController.text.trim().isNotEmpty
                    ? tempMessageType == "Archive"
                        ? true
                        : false
                    : showMessagesOfArchive);

    if (messageService.response_Status_Code_API_6 == 200) {
      setState(() {
        temp = 0;
        var response = messageService.messageDeleteStatus;
        if (response == "true") {
          showToast(
            context,
            message: 'Message(s) Deleted',
            type: ToastificationType.success,
          );
          isEducationMaterialThread = false;
          isMessageDetailScreenVisible = false;
          refreshVariableForMessageAction();
          messageDetailLoading = true;
          searchTextController.text.trim().isNotEmpty
              ? getSearchMessageList(searchTextController.text.trim())
              : isUnreadMessageFilterIconClicked
                  ? getUnreadMessageList()
                  : getMessageList();
        }
      });
    } else {
      if (messageService.response_Status_Code_API_6 != 205) {
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  Future<void> getMessageDetailForAction() async {
    setState(() {
      messageDetailLoading = true;
      htmlContentForForward = "";
      messageMenuPopupVisibility = false;
      messageThreadMenuPopupVisibility = false;
    });
    MessageService messageService = MessageService();
    await messageService.getMessageThread(
        parentMessageId.toString(), tempMessageType);
    if (messageService.response_Status_Code_API_4 == 200) {
      setState(() {
        messageThreadList = messageService.messageThreadModel!;
        if (isReplyMessageScreenVisible) {
          messageThreadList.forEach((element) {
            if (element.isEducationMaterial!) {
              messageDescription +=
                  "From: " + element.fromEmail! + " | " + element.messageTime!;
            } else {
              messageDescription +=
                  "\n\n\n----------------------------------\n" +
                      "From: " +
                      element.fromEmail! +
                      " | " +
                      element.messageTime! +
                      "\n\n" +
                      element.messageText!;
            }
          });
        } else {
          if (messageThreadList[0].isEducationMaterial!)
            htmlContentForForward = messageThreadList[0].messageText!;
          messageThreadList.forEach((element) {
            if (element.isEducationMaterial!) {
              messageDescription +=
                  "As discussed attached is Ref letter from Dr ${messageThreadList[0].fromEmail}.\n\n-----Forward Message-----";
              messageDescription += "\n" +
                  "From: " +
                  element.fromEmail! +
                  " | " +
                  element.messageTime! +
                  "\n";
              messageDescription += "Sending you my Education Material";
            } else {
              messageDescription += "\n\n\n-----Forwarded Message-----\n" +
                  "From: " +
                  element.fromEmail! +
                  " | " +
                  element.messageTime! +
                  "\n\n" +
                  element.messageText!;
            }
          });
        }

        messageSubject = messageThreadList[0].messageSubject!;

        isReplyMessageScreenVisible
            ? messageSubject.startsWith("Re:")
                ? subjectController.text = messageSubject
                : subjectController.text = "Re: " + messageSubject
            : isForwardMessageScreenVisible
                ? messageSubject.startsWith("Fwd:")
                    ? subjectController.text = messageSubject
                    : subjectController.text = "Fwd: " + messageSubject
                : null;

        messageController.text = messageDescription;
        if (isForwardMessageScreenVisible &&
            messageThreadList[messageThreadList.length - 1]
                    .ptMessageAttachment!
                    .length >
                0) {
          messageThreadList[messageThreadList.length - 1]
              .ptMessageAttachment!
              .forEach((element) {
            messageAttachmentList.add(element);
            totalFileAttachmentSize =
                totalFileAttachmentSize + element.attachmentFileSize!;
          });
        }

        messageDetailLoading = false;
        if (subjectController.text.length > 988) {
          _validateSubject = true;
          subjectErrorText = "Max Characters 988";
        } else if (messageController.text.length > 38400) {
          _validateEmailBody = true;
          messageErrorText = "Max Characters 384000";
        }
      });
    } else {
      if (messageService.response_Status_Code_API_4 != 205) {
        setState(() {
          messageDetailLoading = false;
        });
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
    getAttachmentLimit();
  }

        requestModel.isCustomer = true;

        DateFormat dateFormat = DateFormat("yyyy-MM-dd hh:mm");

        MessageForwardModel requestModel = MessageForwardModel();
        requestModel.ptcustomerId = int.parse(GlobalParams.PT_CUSTOMER_ID);
        requestModel.isActive = true;
        requestModel.toEmail = toController.text.trim();
        requestModel.messageSubject = subjectController.text.trim();
        requestModel.messageText =
            messageController.text.trim() + "\n\n" + htmlContentForForward;
        requestModel.messageDateTime = dateFormat.format(currentDate);
        requestModel.messageStatus = "Sent";
        requestModel.isDirectMessage = false;
        requestModel.kno2Id = "";
        requestModel.isCustomer = true;
        requestModel.isIncoming = false;
        requestModel.isFavourite = true;
        requestModel.ptMessageAttachment = messageAttachmentList;
        requestModel.isReplied = false;
        requestModel.parentMessageId = parentMessageId;
        requestModel.ptPracticePersonId = 118;
        requestModel.userId = int.parse(GlobalParams.USER_ID);
        requestModel.switchUserId = int.parse(GlobalParams.SWITCH_USER_ID);

        MessageService messageService = MessageService();
        await messageService.forwardEmail(requestModel);
        if (messageService.response_Status_Code_API_8 == 200) {
          var response = messageService.messageForwardStatus;
          if (response == "true") {
            setState(() {
              isForwardMessageScreenVisible = false;
              isMessageSentScreenVisible = true;
              messageSendingLoading = false;
              refreshVariableForMessageAction();
            });
            searchTextController.text.trim().isNotEmpty
                ? getSearchMessageList(searchTextController.text.trim())
                : getMessageList();
          }
        } else {
          if (messageService.response_Status_Code_API_8 != 205) {
            setState(() {
              messageSendingLoading = false;
            });
            showToast(
              context,
              message:
                  'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
              type: ToastificationType.error,
            );
          }
        }
      } else if (subjectController.text.trim().isEmpty) {
        setState(() {
          _validateSubject = true;
          subjectErrorText = "Subject is required";
        });
      } else if (subjectController.text.trim().isEmpty) {
        setState(() {
          _validateSubject = true;
          subjectErrorText = "Subject is required";
        });
      } else if (subjectController.text.length > 988) {
        _validateSubject = true;
        subjectErrorText = "Max Characters 988";
      } else if (messageController.text.length > 384000) {
        _validateEmailBody = true;
        messageErrorText = "Max Characters 384000";
      } else if (toController.text.isEmpty) {
        setState(() {
          _validateToEmail = true;
          _invalidEmailMessage = 'Email is required';
        });
      }
    }
  }

  Future<void> getMessageAttachmentById(
      int attachmentId, String? fileName) async {
    await messageService.getMessageAttachmentById(attachmentId);
    if (messageService.response_Status_Code_API_11 == 200) {
      downloadFile(messageService.messageAttachmentBase64, fileName);
    } else {
      if (messageService.response_Status_Code_API_11 != 205) {
        showToast(
          context,
          message:
              'An unexpected error has occurred. Please try again later. If the problem persists, call our office.',
          type: ToastificationType.error,
        );
      }
    }
  }

  downloadFile(String? base64File, String? fileName) {
    final blob = html.Blob([base64Decode(base64File!)]);
    final url = html.Url.createObjectUrlFromBlob(blob);
    final anchor = html.document.createElement('a') as html.AnchorElement
      ..href = url
      ..style.display = 'none'
      ..download = fileName;
    html.document.body!.children.add(anchor);
    anchor.click();
    html.document.body!.children.remove(anchor);
    html.Url.revokeObjectUrl(url);
  }

  convertFileToBase64(PlatformFile file) {
    if (file.extension == 'jpg' ||
        file.extension == 'pdf' ||
        file.extension == 'doc' ||
        file.extension == 'png' ||
        file.extension == 'txt' ||
        file.extension == 'xml' ||
        file.extension == 'JPG' ||
        file.extension == 'PDF' ||
        file.extension == 'DOC' ||
        file.extension == 'PNG' ||
        file.extension == 'TXT' ||
        file.extension == 'XML') {
      String file64 = base64Encode(file.bytes!);
      MessageAttachment attachment = MessageAttachment();
      attachment.attachmentId = 0;
      attachment.attachmentType = file.extension;
      attachment.attachmentBase64FileContent = file64;
      attachment.attachmentFileName = file.name;
      attachment.attachmentFileSize = fileSizeConversion(file);
      setState(() {
        messageAttachmentList.add(attachment);
      });
    } else {
      showToast(
        context,
        message: 'Invalid File Format of ${file.name.split('.')[0]}',
        type: ToastificationType.warning,
      );
    }
  }

  fileSizeConversion(PlatformFile file) {
    return ((file.size) / 1024).toDouble();
  }

  checkFileSize(PlatformFile file) {
    final size = file.size / 1024;
    if (size <= 25600 && totalFileAttachmentSize + size <= 25600) {
      setState(() {
        totalFileAttachmentSize = totalFileAttachmentSize + size;
      });
      return true;
    }
    return false;
  }

  checkFileSizeIsZero(PlatformFile file) {
    final size = file.size / 1024;
    if (size > 0) {
      return false;
    }
    return true;
  }

  void loadSelectedFiles(List<PlatformFile> file) {
    if (file.length > int.parse(attachmentLimit) ||
        messageAttachmentList.length + file.length >
            int.parse(attachmentLimit)) {
      showToast(
        context,
        message: 'Maximum $attachmentLimit attachments are allowed',
        type: ToastificationType.warning,
      );
    } else {
      for (int i = 0; i < file.length; i++) {
        bool isDuplicate = messageAttachmentList.any(
            (existingFile) => existingFile.attachmentFileName == file[i].name);
        if (isDuplicate) {
          showToast(
            context,
            message: 'This file has already been uploaded',
            type: ToastificationType.warning,
          );
        } else if (checkFileSizeIsZero(file[i])) {
          if (!attachmentSizeIsZeroToastVisibility) {
            showToast(
              context,
              message: 'Attached files must be greater than 0 bytes!',
              type: ToastificationType.warning,
            );
            setState(() {
              attachmentSizeIsZeroToastVisibility = true;
            });
          }
        } else {
          if (checkFileSize(file[i])) {
            convertFileToBase64(file[i]);
          } else {
            if (!attachmentSizeIsMaxToastVisibility) {
              showToast(
                context,
                message:
                    'The attachment size exceeds the allowable limit. Maximum size of all attachments allowed is 25 MB.',
                type: ToastificationType.warning,
              );
            }
            setState(() {
              attachmentSizeIsMaxToastVisibility = true;
            });
          }
        }
      }
      setState(() {
        attachmentSizeIsMaxToastVisibility = false;
        attachmentSizeIsZeroToastVisibility = false;
      });
    }
  }

  Widget messageTile(int index, MessageListModel messageData) {
    selectedMessageIdsForAction!.isEmpty ? messageData.isChecked = false : null;
    return ListTile(
      hoverColor: ThemeSelection.listTileHoverColor,
      tileColor: ThemeSelection.listTileDefaultColor,
      selectedTileColor: selectedMessageIndex == index &&
              !Responsive.isMobile(context) &&
              !isComposeMessageScreenVisible &&
              !isMessageSentScreenVisible
          ? ThemeSelection.listTileSelectedColor
          : ThemeSelection.listTileDefaultColor,
      selected: selectedMessageIndex == index ? true : false,
      onTap: () {
        setState(() {
          selectedMessageIdsForAction!.clear();
          selectedMessageIndex = index;
          messageData.isMessageSeen = true;
          parentMessageId = messageData.messageId!;
          messageType = messageData.messageType!;
          tempMessageType = messageData.tempMessageType!;
          getMessageThread();
          if (Responsive.isMobile(context)) {
            isMessageDetailScreenVisible = true;
          } else {
            isComposeMessageScreenVisible = false;
            isForwardMessageScreenVisible = false;
            isReplyMessageScreenVisible = false;
            isMessageSentScreenVisible = false;
            isMessageDetailScreenVisible = true;
          }
        });
      },
      onLongPress: Responsive.isMobile(context)
          ? () {
              setState(() {
                messageData.isChecked = !messageData.isChecked!;
                if (selectedMessageIdsForAction!
                    .contains(messageData.messageId.toString())) {
                  selectedMessageIdsForAction!
                      .remove(messageData.messageId.toString());
                } else {
                  selectedMessageIdsForAction!
                      .add(messageData.messageId.toString());
                }
              });
            }
          : null,
      contentPadding: EdgeInsets.only(left: 2.0, right: 2.0),
      minLeadingWidth: searchTextController.text.isEmpty ? 40 : 0,
      leading: Padding(
        padding: const EdgeInsets.only(left: 10.0),
        child: searchTextController.text.isEmpty
            ? GestureDetector(
                onTap: () {
                  setState(() {
                    messageData.isChecked = !messageData.isChecked!;
                    if (selectedMessageIdsForAction!
                        .contains(messageData.messageId.toString())) {
                      selectedMessageIdsForAction!
                          .remove(messageData.messageId.toString());
                    } else {
                      selectedMessageIdsForAction!
                          .add(messageData.messageId.toString());
                    }
                  });
                },
                child: AnimatedSwitcher(
                  switchInCurve: Curves.bounceIn,
                  switchOutCurve: Curves.bounceOut,
                  duration: Duration(milliseconds: 1),
                  child: Responsive.isMobile(context)
                      ? messageData.isChecked!
                          ? CircleAvatar(
                              backgroundColor: Colors.green,
                              child: Center(
                                child: Icon(Icons.check),
                              ),
                            )
                          : messageData.messageStatus == "Sent"
                              ? GlobalParams.USER_PROFILE_PHOTO_BASE64 == null
                                  ? CircleAvatar(
                                      backgroundColor: ThemeSelection.BGColor,
                                      child: Center(
                                        child: Text(
                                          messageData.userInitials!,
                                          style: CommonTextStyle.of(context)
                                              .H22_SUR(ThemeSelection
                                                  .secondaryTextColor),
                                        ),
                                      ),
                                    )
                                  : CircleAvatar(
                                      backgroundImage: MemoryImage(
                                        GlobalParams.USER_PROFILE_PHOTO_BASE64!,
                                        scale: Responsive.isMobile(context)
                                            ? 2
                                            : Responsive.isTablet(context)
                                                ? 2
                                                : 1,
                                      ),
                                    )
                              : CircleAvatar(
                                  backgroundColor: ThemeSelection.BGColor,
                                  child: Center(
                                    child: Text(
                                      messageData.userInitials!,
                                      style: CommonTextStyle.of(context)
                                          .H22_SUR(ThemeSelection
                                              .secondaryTextColor),
                                    ),
                                  ),
                                )
                      : messageData.isChecked!
                          ? Icon(
                              Icons.check_box,
                              color: Colors.green,
                            )
