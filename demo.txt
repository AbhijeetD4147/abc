import 'dart:convert';

import 'package:first_insight/utils/theme_selection_service.dart';
import 'package:first_insight/utils/appBar.dart';
import 'package:first_insight/utils/base_widget.dart';
import 'package:first_insight/utils/constants.dart';
import 'package:first_insight/utils/responsive.dart';
import 'package:flutter/material.dart';
import 'package:get/get.dart';
import 'package:material_symbols_icons/symbols.dart';
import 'package:sizer/sizer.dart';
import 'package:universal_html/html.dart' as html;
import 'package:webviewx/webviewx.dart';

class PoliciesFormScreen extends StatefulWidget {
  final String? formName;
  final String? formURL;
  final String? formStatus;
  final String? formBase64;

  const PoliciesFormScreen({
    super.key,
    this.formName,
    this.formURL,
    this.formStatus,
    this.formBase64,
  });

  @override
  State<PoliciesFormScreen> createState() => _PoliciesFormScreenState();
}

class _PoliciesFormScreenState extends State<PoliciesFormScreen>
    with TickerProviderStateMixin {
  @override
  Widget build(BuildContext context) {
    return GestureDetector(
      onTap: () {
        setState(() {
          ApiPath.isSettingIconPopupVisible = false;
          ApiPath.isSettingIconPopupOpen.value = false;
        });
      },
      child: Scaffold(
        appBar: !Responsive.isMobile(context)
            ? CommonAppBar(
                context,
                TabController(length: 6, vsync: this, initialIndex: 2),
                setState,
                2,
                "")
            : PreferredSize(
                preferredSize: Size(0, 0),
                child: SizedBox(),
              ),
        body: Responsive(
          mobile: commonView(),
          tablet: commonView(),
          desktop: commonView(),
          k4Desktop: commonView(),
        ),
      ),
    );
  }

  downloadFile(String base64File, String fileName) {
    final blob = html.Blob([base64Decode(base64File)]);
    final url = html.Url.createObjectUrlFromBlob(blob);
    final anchor = html.document.createElement('a') as html.AnchorElement
      ..href = url
      ..style.display = 'none'
      ..download = fileName;
    html.document.body!.children.add(anchor);

    anchor.click();

    html.document.body!.children.remove(anchor);
    html.Url.revokeObjectUrl(url);
  }

  Widget commonView() {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Container(
          color: Responsive.isMobile(context)
              ? ThemeSelection.BGColor
              : Colors.transparent,
          padding: EdgeInsets.symmetric(horizontal: 1.w, vertical: 1.h),
          child: Row(
            mainAxisSize: MainAxisSize.max,
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: [
              customIconButton(
                context,
                onPressed: () {
                  Get.back();
                },
                splashColor: ThemeSelection.iconPressedSecondaryColor,
                hoverColor: ThemeSelection.iconHoverSecondaryColor,
                iconData: Symbols.arrow_back_ios_new,
                iconColor: Responsive.isMobile(context)
                    ? ThemeSelection.iconDefaultTertiaryColor
                    : ThemeSelection.iconDefaultSecondaryColor,
              ),
              Text(
                widget.formName!,
                style: TextStyle(
                    fontFamily: 'SegoeUi',
                    fontSize: 20.0,
                    fontWeight: FontWeight.bold,
                    color: Responsive.isMobile(context)
                        ? ThemeSelection.secondaryTextColor
                        : ThemeSelection.primaryTextColor),
              ),
              Visibility(
                visible: widget.formStatus == 'Accepted',
                child: customIconButtonWithTooltip(
                  context,
                  toolTipMessage: 'Download PDF',
                  splashColor: ThemeSelection.iconPressedSecondaryColor,
                  hoverColor: ThemeSelection.iconHoverSecondaryColor,
                  iconColor: Responsive.isMobile(context)
                      ? ThemeSelection.iconDefaultTertiaryColor
                      : ThemeSelection.iconDefaultSecondaryColor,
                  iconData: Symbols.download,
                  onPressed: () {
                    downloadFile(widget.formBase64!, widget.formName!);
                  },
                ),
              ),
            ],
          ),
        ),
        Divider(
          height: 1,
          color: ThemeSelection.greyBorderColor,
        ),
        WebViewX(
          height: 90.h,
          initialContent: '<center>Something Went Wrong!</center>',
          initialSourceType: SourceType.HTML,
          onPageStarted: (src) {},
          onWebViewCreated: (webViewController) async {
            webViewController.loadContent(
              widget.formURL!,
              SourceType.URL,
            );
          },
        ),
      ],
    );
  }
}
